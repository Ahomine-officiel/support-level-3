    78→/// (fichier texture sans extension, force émissive du matériau).
    79→pub fn material_def(name: &str) -> (String, f32) {
    80→    let strength = match name {
    81→        "led_strip" => 2.0,
    82→        "screen_on" => 0.8,
    83→        "terminal_screen" => 0.9,
    84→        "exit_sign" => 1.4,
    85→        "eyes" => 2.6,
    86→        "light_panel" => 1.25,
    87→        "battery" => 0.35,
    88→        "receipt_paper" => 0.05,
    89→        "poster_a" | "poster_b" | "poster_c" | "poster_d" => 0.04,
    90→        "entity_mask" => 0.03,
    91→        _ => 0.0,
    92→    };
    93→    (name.to_string(), strength)
    94→}

   146→    pub fn new(window: std::sync::Arc<Window>) -> Renderer {
   147→        let size = window.inner_size();
   148→        let instance = wgpu::Instance::new(wgpu::InstanceDescriptor::default());
   149→        let surface = instance
   150→            .create_surface(window.clone())
   151→            .expect("création surface");
   152→        let adapter = pollster::block_on(instance.request_adapter(&wgpu::RequestAdapterOptions {
   153→            power_preference: wgpu::PowerPreference::HighPerformance,
   154→            compatible_surface: Some(&surface),
   155→            force_fallback_adapter: false,
   156→        }))

   161→                label: Some("sl3-device"),

   170→        let caps = surface.get_capabilities(&adapter);
   171→        let format = caps
   172→            .formats
   173→            .iter()
   174→            .find(|f| f.is_srgb())
   175→            .copied()

   178→            usage: wgpu::TextureUsages::RENDER_ATTACHMENT,
   179→            format,
   180→            width: size.width.max(1),
   181→            height: size.height.max(1),
   182→            present_mode: wgpu::PresentMode::Fifo,
   183→            alpha_mode: caps.alpha_modes[0],
   184→            view_formats: vec![],
   185→            desired_maximum_frame_latency: 2,
   186→        };
   187→        surface.configure(&device, &surface_config);
   188→
   189→        // Modèles + matériaux.
   190→        let models = model::load_all_models(&device);
   191→        let mut material_bind_groups = HashMap::new();
   192→        let mat_layout = Self::material_layout(&device);
   193→        for name in [
   194→            "concrete", "concrete_dark", "floor_hall", "floor_corridor", "floor_office",
   195→            "floor_server", "floor_arch", "floor_elec", "ceiling", "metal_dark",
   196→            "rack_front", "server_front", "led_strip", "screen_off", "screen_on",
   197→            "terminal_screen", "desk_wood", "terminal_body", "chair_fabric", "door_metal",
   198→            "door_frame", "shelf_metal", "cardboard", "entity_cloth", "entity_mask",
   199→            "eyes", "tech_vest", "tech_pants", "tech_head", "receipt_paper",
   200→            "breaker_panel", "exit_sign", "battery", "extinguisher", "duct",
   201→            "light_panel", "hazard", "poster_a", "poster_b", "poster_c", "poster_d",
   202→        ] {
   203→            let (file, strength) = material_def(name);
   204→            let path = Path::new("assets/textures").join(format!("{file}.png"));
   205→            let tex = texture::load_png(&device, &queue, &path);
   206→            let mat_buf = device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
   207→                label: None,
   208→                contents: bytemuck::cast_slice(&[1.0f32, 1.0, 1.0, strength]),
   209→                usage: wgpu::BufferUsages::UNIFORM | wgpu::BufferUsages::COPY_DST,
   210→            });
   211→            let bg = device.create_bind_group(&wgpu::BindGroupDescriptor {
   212→                label: Some(name),
   213→                layout: &mat_layout,
   214→                entries: &[
   215→                    wgpu::BindGroupEntry { binding: 0, resource: wgpu::BindingResource::Sampler(&Self::world_sampler(&device)) },
   216→                    wgpu::BindGroupEntry { binding: 1, resource: wgpu::BindingResource::TextureView(&tex.view) },
   217→                    wgpu::BindGroupEntry { binding: 2, resource: mat_buf.as_entire_binding() },
   218→                ],
   219→            });
   220→            material_bind_groups.insert(name.to_string(), bg);
   221→        }
   222→
   223→        // Uniforms monde.
   224→        let world_uniform_buf = device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
   225→            label: Some("world-uniform"),
   226→            contents: bytemuck::bytes_of(&WorldUniform {
   227→                view_proj: glam::Mat4::IDENTITY.to_cols_array_2d(),
   228→                cam_pos: [0.0; 4],
   229→                light_pos: [[0.0; 4]; MAX_LIGHTS],
   230→                light_col: [[0.0; 4]; MAX_LIGHTS],
   231→                flash_pos: [0.0; 4],
   232→                flash_dir: [0.0; 4],
   233→                misc: [0.0; 4],
   234→                flash_col: [0.0; 4],
   235→            }),
   236→            usage: wgpu::BufferUsages::UNIFORM | wgpu::BufferUsages::COPY_DST,
   237→        });
   238→        let world_bind_layout = Self::world_bind_layout(&device);
   239→        let world_bind0 = device.create_bind_group(&wgpu::BindGroupDescriptor {
   240→            label: Some("world-bind0"),
   241→            layout: &world_bind_layout,
   242→            entries: &[wgpu::BindGroupEntry {
   243→                binding: 0,
   244→                resource: world_uniform_buf.as_entire_binding(),
   245→            }],
   246→        });
   247→
   248→        let world_shader = device.create_shader_module(wgpu::ShaderModuleDescriptor {
   249→            label: Some("world.wgsl"),
   250→            source: wgpu::ShaderSource::Wgsl(include_str!("../shaders/world.wgsl").into()),
   251→        });
   252→        let world_pipeline = Self::world_pipeline(&device, &world_bind_layout, &mat_layout, &world_shader, format);
   253→
   254→        // Post-process.
   255→        let post_shader = device.create_shader_module(wgpu::ShaderModuleDescriptor {
   256→            label: Some("post.wgsl"),
   257→            source: wgpu::ShaderSource::Wgsl(include_str!("../shaders/post.wgsl").into()),
   258→        });
   259→        let post_layout = device.create_bind_group_layout(&wgpu::BindGroupLayoutDescriptor {
   260→            label: Some("post-layout"),
   261→            entries: &[
   262→                wgpu::BindGroupLayoutEntry { binding: 0, visibility: wgpu::ShaderStages::FRAGMENT, ty: wgpu::BindingType::Buffer { ty: wgpu::BufferBindingType::Uniform, has_dynamic_offset: false, min_binding_size: None }, count: None },
   263→                wgpu::BindGroupLayoutEntry { binding: 1, visibility: wgpu::ShaderStages::FRAGMENT, ty: wgpu::BindingType::Sampler(wgpu::SamplerBindingType::Filtering), count: None },
   264→                wgpu::BindGroupLayoutEntry { binding: 2, visibility: wgpu::ShaderStages::FRAGMENT, ty: wgpu::BindingType::Texture { sample_type: wgpu::TextureSampleType::Float { filterable: true }, multisampled: false, view_dimension: wgpu::TextureViewDimension::D2 }, count: None },
   265→            ],
   266→        });
   267→        let post_uniform_buf = device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
   268→            label: Some("post-uniform"),
   269→            contents: bytemuck::cast_slice(&[0.0f32; 4]),
   270→            usage: wgpu::BufferUsages::UNIFORM | wgpu::BufferUsages::COPY_DST,
   271→        });
   272→
   273→        let (offscreen_tex, offscreen_view) = Self::make_offscreen(&device, size.width.max(1), size.height.max(1), format);
   274→        let (depth_tex, depth_view) = Self::make_depth(&device, size.width.max(1), size.height.max(1));
   275→
   276→        let post_bind = device.create_bind_group(&wgpu::BindGroupDescriptor {
   277→            label: Some("post-bind"),
   278→            layout: &post_layout,
   279→            entries: &[
   280→                wgpu::BindGroupEntry { binding: 0, resource: post_uniform_buf.as_entire_binding() },
   281→                wgpu::BindGroupEntry { binding: 1, resource: wgpu::BindingResource::Sampler(&texture::linear_sampler(&device, false)) },
   282→                wgpu::BindGroupEntry { binding: 2, resource: wgpu::BindingResource::TextureView(&offscreen_view) },
   283→            ],
   284→        });
   285→        let post_pipeline = Self::post_pipeline(&device, &post_layout, &post_shader, format);
   286→
   287→        // UI.
   288→        let ui_shader = device.create_shader_module(wgpu::ShaderModuleDescriptor {
   289→            label: Some("ui.wgsl"),
   290→            source: wgpu::ShaderSource::Wgsl(include_str!("../shaders/ui.wgsl").into()),
   291→        });
   292→        let ui_layout0 = device.create_bind_group_layout(&wgpu::BindGroupLayoutDescriptor {
   293→            label: Some("ui-layout0"),
   294→            entries: &[wgpu::BindGroupLayoutEntry {
   295→                binding: 0,
   296→                visibility: wgpu::ShaderStages::VERTEX,
   297→                ty: wgpu::BindingType::Buffer { ty: wgpu::BufferBindingType::Uniform, has_dynamic_offset: false, min_binding_size: None },
   298→                count: None,
   299→            }],
   300→        });
   301→        let ui_layout1 = device.create_bind_group_layout(&wgpu::BindGroupLayoutDescriptor {
   302→            label: Some("ui-layout1"),
   303→            entries: &[
   304→                wgpu::BindGroupLayoutEntry { binding: 0, visibility: wgpu::ShaderStages::FRAGMENT, ty: wgpu::BindingType::Sampler(wgpu::SamplerBindingType::Filtering), count: None },
   305→                wgpu::BindGroupLayoutEntry { binding: 1, visibility: wgpu::ShaderStages::FRAGMENT, ty: wgpu::BindingType::Texture { sample_type: wgpu::TextureSampleType::Float { filterable: true }, multisampled: false, view_dimension: wgpu::TextureViewDimension::D2 }, count: None },
   306→            ],
   307→        });
   308→        let ui_uniform_buf = device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
   309→            label: Some("ui-uniform"),
   310→            contents: bytemuck::cast_slice(&[size.width as f32, size.height as f32, 0.0, 0.0]),
   311→            usage: wgpu::BufferUsages::UNIFORM | wgpu::BufferUsages::COPY_DST,
   312→        });
   313→        let ui_bind0 = device.create_bind_group(&wgpu::BindGroupDescriptor {
   314→            label: Some("ui-bind0"),
   315→            layout: &ui_layout0,
   316→            entries: &[wgpu::BindGroupEntry { binding: 0, resource: ui_uniform_buf.as_entire_binding() }],
   317→        });
   318→        let white_tex = texture::white_texture(&device, &queue);
   319→        let white_bg = device.create_bind_group(&wgpu::BindGroupDescriptor {
   320→            label: Some("ui-white"),
   321→            layout: &ui_layout1,
   322→            entries: &[
   323→                wgpu::BindGroupEntry { binding: 0, resource: wgpu::BindingResource::Sampler(&texture::linear_sampler(&device, false)) },
   324→                wgpu::BindGroupEntry { binding: 1, resource: wgpu::BindingResource::TextureView(&white_tex.view) },
   325→            ],
   326→        });
   327→        let (font_tex, font) = ui::load_font(&device, &queue);
   328→        let font_bg = device.create_bind_group(&wgpu::BindGroupDescriptor {
   329→            label: Some("ui-font"),
   330→            layout: &ui_layout1,
   331→            entries: &[
   332→                wgpu::BindGroupEntry { binding: 0, resource: wgpu::BindingResource::Sampler(&texture::linear_sampler(&device, false)) },
   333→                wgpu::BindGroupEntry { binding: 1, resource: wgpu::BindingResource::TextureView(&font_tex.view) },
   334→            ],
   335→        });
   336→        let ui_pipeline = Self::ui_pipeline(&device, &ui_layout0, &ui_layout1, &ui_shader, format);
   337→        let ui_buf = device.create_buffer(&wgpu::BufferDescriptor {
   338→            label: Some("ui-verts"),
   339→            size: (4 * 4096 * std::mem::size_of::<ui::UiVertex>() as u64).max(64),
   340→            usage: wgpu::BufferUsages::VERTEX | wgpu::BufferUsages::COPY_DST,
   341→            mapped_at_creation: false,
   342→        });
   343→
   344→        Renderer {
   345→            device,
   346→            queue,
   347→            surface,
   348→            surface_config,
   349→            window,
   350→            models,
   351→            world_pipeline,
   352→            world_bind0,
   353→            world_uniform_buf,
   354→            material_bind_groups,
   355→            post_pipeline,
   356→            post_bind,

   358→            offscreen_view,
   359→            offscreen_tex,
   360→            depth_view,
   361→            depth_tex,
   362→            ui_pipeline,
   363→            ui_bind0,
   364→            ui_uniform_buf,
   365→            font_bg,
   366→            white_bg,
   367→            font,
   368→            ui_buf,
   369→            dyn_scratch: HashMap::new(),
   370→        }

   401→    fn world_pipeline(
   402→        device: &wgpu::Device,
   403→        bind0: &wgpu::BindGroupLayout,
   404→        bind1: &wgpu::BindGroupLayout,
   405→        shader: &wgpu::ShaderModule,
   406→        format: wgpu::TextureFormat,
   407→    ) -> wgpu::RenderPipeline {
   408→        let layout = device.create_pipeline_layout(&wgpu::PipelineLayoutDescriptor {
   409→            label: Some("world-pll"),
   410→            bind_group_layouts: &[bind0, bind1],
   411→            push_constant_ranges: &[],
   412→        });
   413→        device.create_render_pipeline(&wgpu::RenderPipelineDescriptor {
   414→            label: Some("world-pipeline"),
   415→            layout: Some(&layout),
   416→            vertex: wgpu::VertexState {
   417→                module: shader,
   418→                entry_point: "vs",
   419→                buffers: &[Vertex::LAYOUT, INSTANCE_LAYOUT],
   420→                compilation_options: Default::default(),
   421→            },
   422→            fragment: Some(wgpu::FragmentState {
   423→                module: shader,
   424→                entry_point: "fs",
   425→                targets: &[Some(wgpu::ColorTargetState {
   426→                    format,
   427→                    blend: None,
   428→                    write_mask: wgpu::ColorWrites::ALL,
   429→                })],
   430→                compilation_options: Default::default(),
   431→            }),
   432→            primitive: wgpu::PrimitiveState::default(),
   433→            depth_stencil: Some(wgpu::DepthStencilState {
   434→                format: wgpu::TextureFormat::Depth24Plus,
   435→                depth_write_enabled: true,
   436→                depth_compare: wgpu::CompareFunction::Less,
   437→                stencil: Default::default(),
   438→                bias: Default::default(),
   439→            }),
   440→            multisample: Default::default(),
   441→            multiview: None,
   442→            cache: None,
   443→        })
   444→    }

   479→    fn ui_pipeline(device: &wgpu::Device, l0: &wgpu::BindGroupLayout, l1: &wgpu::BindGroupLayout, shader: &wgpu::ShaderModule, format: wgpu::TextureFormat) -> wgpu::RenderPipeline {
   480→        use ui::UiVertex;
   481→        let pll = device.create_pipeline_layout(&wgpu::PipelineLayoutDescriptor {
   482→            label: Some("ui-pll"),
   483→            bind_group_layouts: &[l0, l1],
   484→            push_constant_ranges: &[],
   485→        });
   486→        device.create_render_pipeline(&wgpu::RenderPipelineDescriptor {
   487→            label: Some("ui-pipeline"),
   488→            layout: Some(&pll),
   489→            vertex: wgpu::VertexState {
   490→                module: shader,
   491→                entry_point: "vs",
   492→                buffers: &[UiVertex::LAYOUT],
   493→                compilation_options: Default::default(),
   494→            },
   495→            fragment: Some(wgpu::FragmentState {
   496→                module: shader,
   497→                entry_point: "fs",
   498→                targets: &[Some(wgpu::ColorTargetState {
   499→                    format,
   500→                    blend: Some(wgpu::BlendState::ALPHA_BLENDING),
   501→                    write_mask: wgpu::ColorWrites::ALL,
   502→                })],
   503→                compilation_options: Default::default(),
   504→            }),
   505→            primitive: wgpu::PrimitiveState { cull_mode: None, ..Default::default() },
   506→            depth_stencil: None,
   507→            multisample: Default::default(),
   508→            multiview: None,
   509→            cache: None,
   510→        })
   511→    }

   547→    pub fn resize(&mut self, w: u32, h: u32) {
   548→        let (w, h) = (w.max(1), h.max(1));
   549→        self.surface_config.width = w;
   550→        self.surface_config.height = h;
   551→        self.surface.configure(&self.device, &self.surface_config);
   552→        let (t, v) = Self::make_offscreen(&self.device, w, h, self.surface_config.format);
   553→        self.offscreen_tex = t;
   554→        self.offscreen_view = v;
   555→        let post_layout = self.post_pipeline.get_bind_group_layout(0);
   556→        self.post_bind = self.device.create_bind_group(&wgpu::BindGroupDescriptor {
   557→            label: Some("post-bind"),
   558→            layout: &post_layout,
   559→            entries: &[
   560→                wgpu::BindGroupEntry { binding: 0, resource: self.post_uniform_buf.as_entire_binding() },
   561→                wgpu::BindGroupEntry { binding: 1, resource: wgpu::BindingResource::Sampler(&texture::linear_sampler(&self.device, false)) },
   562→                wgpu::BindGroupEntry { binding: 2, resource: wgpu::BindingResource::TextureView(&self.offscreen_view) },
   563→            ],
   564→        });
   565→        let (dt, dv) = Self::make_depth(&self.device, w, h);
   566→        self.depth_tex = dt;
   567→        self.depth_view = dv;
   568→        let ui_data = [w as f32, h as f32, 0.0, 0.0];
   569→        self.queue.write_buffer(&self.ui_uniform_buf, 0, bytemuck::cast_slice(&ui_data));
   570→    }

   572→    // ---------- construction des batchs statiques ----------
   573→    pub fn build_static(&self, insts: &[(String, InstanceData)]) -> StaticBatches {
   574→        let mut groups: HashMap<PartKey, Vec<InstanceRaw>> = HashMap::new();
   575→        for (model_name, inst) in insts {
   576→            if let Some(model) = self.models.get(model_name) {
   577→                for (pi, part) in model.parts.iter().enumerate() {
   578→                    groups
   579→                        .entry(PartKey { model: model_name.clone(), part: pi })
   580→                        .or_default()
   581→                        .push(inst.raw());
   582→                    let _ = &part.mat;
   583→                }
   584→            }
   585→        }
   586→        let mut batches = Vec::new();
   587→        let mut total = 0usize;
   588→        for (key, raws) in groups {
   589→            let buffer = self.device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
   590→                label: Some("static-instances"),
   591→                contents: bytemuck::cast_slice(&raws),
   592→                usage: wgpu::BufferUsages::VERTEX | wgpu::BufferUsages::COPY_DST,
   593→            });
   594→            total += raws.len();
   595→            batches.push(StaticBatch { key, buffer, count: raws.len() as u32 });
   596→        }
   597→        StaticBatches { batches, total_instances: total }
   598→    }

   600→    // ---------- dessin ----------
   601→    pub fn draw_instances_pass(
   602→        &mut self,
   603→        pass: &mut wgpu::RenderPass<'static>,
   604→        statics: &StaticBatches,
   605→        dynamics: &[(String, Vec<InstanceData>)],
   606→    ) {
   607→        pass.set_pipeline(&self.world_pipeline);
   608→        pass.set_bind_group(0, &self.world_bind0, &[]);
   609→
   610→        // Statiques.
   611→        for batch in &statics.batches {
   612→            self.draw_part(pass, &batch.key, &batch.buffer, batch.count);
   613→        }
   614→
   615→        // Dynamiques.
   616→        for (model_name, insts) in dynamics {
   617→            let Some(model) = self.models.get(model_name) else { continue };
   618→            for (pi, _part) in model.parts.iter().enumerate() {
   619→                if insts.is_empty() {
   620→                    continue;
   621→                }
   622→                let key = PartKey { model: model_name.clone(), part: pi };
   623→                let raws: Vec<InstanceRaw> = insts.iter().map(|i| i.raw()).collect();
   624→                let bytes_needed = (raws.len() * std::mem::size_of::<InstanceRaw>()) as u64;
   625→                let needs_grow = self
   626→                    .dyn_scratch
   627→                    .get(&key)
   628→                    .map(|(_, cap)| *cap < bytes_needed)
   629→                    .unwrap_or(true);
   630→                if needs_grow {
   631→                    let cap = bytes_needed.max(1024);
   632→                    let buf = self.device.create_buffer(&wgpu::BufferDescriptor {
   633→                        label: Some("dyn-instances"),
   634→                        size: cap,
   635→                        usage: wgpu::BufferUsages::VERTEX | wgpu::BufferUsages::COPY_DST,
   636→                        mapped_at_creation: false,
   637→                    });
   638→                    self.dyn_scratch.insert(key.clone(), (buf, cap));
   639→                }
   640→                let (buf, _cap) = &self.dyn_scratch[&key];
   641→                self.queue.write_buffer(buf, 0, bytemuck::cast_slice(&raws));
   642→                self.draw_part(pass, &key, buf, raws.len() as u32);
   643→            }
   644→        }
   645→    }
   646→
   647→    fn draw_part(
   648→        &self,
   649→        pass: &mut wgpu::RenderPass<'static>,
   650→        key: &PartKey,
   651→        instance_buf: &wgpu::Buffer,
   652→        count: u32,
   653→    ) {
   654→        let Some(model) = self.models.get(&key.model) else { return };
   655→        let Some(part) = model.parts.get(key.part) else { return };
   656→        let Some(bg) = self.material_bind_groups.get(&part.mat) else { return };
   657→        pass.set_bind_group(1, bg, &[]);
   658→        pass.set_vertex_buffer(0, part.vertex_buf.slice(..));
   659→        pass.set_vertex_buffer(1, instance_buf.slice(..));
   660→        pass.set_index_buffer(part.index_buf.slice(..), wgpu::IndexFormat::Uint32);
   661→        pass.draw_indexed(0..part.count, 0, 0..count);
   662→    }

   664→    #[allow(clippy::too_many_arguments)]
   665→    pub fn render(
   666→        &mut self,
   667→        world_uniform: &WorldUniform,
   668→        statics: &StaticBatches,
   669→        dynamics: &[(String, Vec<InstanceData>)],
   670→        ui_ops: &[ui::UiOp],
   671→        post_params: [f32; 4],
   672→    ) -> Result<(), wgpu::SurfaceError> {
   673→        let frame = self.surface.get_current_texture()?;
   674→        let frame_view = frame.texture.create_view(&wgpu::TextureViewDescriptor::default());
   675→
   676→        self.queue
   677→            .write_buffer(&self.world_uniform_buf, 0, bytemuck::bytes_of(world_uniform));
   678→        self.queue
   679→            .write_buffer(&self.post_uniform_buf, 0, bytemuck::cast_slice(&post_params));
   680→
   681→        // Quads UI.
   682→        let (verts, draws) = ui::build_ui_verts(ui_ops, &self.font);
   683→        if !verts.is_empty() {
   684→            let bytes_needed = (verts.len() * std::mem::size_of::<ui::UiVertex>()) as u64;
   685→            if self.ui_buf.size() < bytes_needed {
   686→                self.ui_buf = self.device.create_buffer(&wgpu::BufferDescriptor {
   687→                    label: Some("ui-verts"),
   688→                    size: bytes_needed * 2,
   689→                    usage: wgpu::BufferUsages::VERTEX | wgpu::BufferUsages::COPY_DST,
   690→                    mapped_at_creation: false,
   691→                });
   692→            }
   693→            self.queue.write_buffer(&self.ui_buf, 0, bytemuck::cast_slice(&verts));
   694→        }
   695→
   696→        let mut encoder = self.device.create_command_encoder(&wgpu::CommandEncoderDescriptor {
   697→            label: Some("frame"),
   698→        });
   699→
   700→        // 1) Monde -> offscreen.
   701→        {
   702→            let pass = encoder.begin_render_pass(&wgpu::RenderPassDescriptor {
   703→                label: Some("world-pass"),
   704→                color_attachments: &[Some(wgpu::RenderPassColorAttachment {
   705→                    view: &self.offscreen_view,
   706→                    resolve_target: None,
   707→                    ops: wgpu::Operations {
   708→                        load: wgpu::LoadOp::Clear(wgpu::Color {
   709→                            r: 0.004,
   710→                            g: 0.005,
   711→                            b: 0.008,
   712→                            a: 1.0,
   713→                        }),
   714→                        store: wgpu::StoreOp::Store,
   715→                    },
   716→                })],
   717→                depth_stencil_attachment: Some(wgpu::RenderPassDepthStencilAttachment {
   718→                    view: &self.depth_view,
   719→                    depth_ops: Some(wgpu::Operations {
   720→                        load: wgpu::LoadOp::Clear(1.0),
   721→                        store: wgpu::StoreOp::Store,
   722→                    }),
   723→                    stencil_ops: None,
   724→                }),
   725→                timestamp_writes: None,
   726→                occlusion_query_set: None,
   727→            });
   728→            // 'pass borrow self...' : étendre la durée de vie via transmute sûr ici
   729→            let mut pass = pass.forget_lifetime();
   730→            self.draw_instances_pass(&mut pass, statics, dynamics);
   731→        }
   732→
   733→        // 2) Post-process -> surface.
   734→        {
   735→            let mut pass = encoder.begin_render_pass(&wgpu::RenderPassDescriptor {
   736→                label: Some("post-pass"),
   737→                color_attachments: &[Some(wgpu::RenderPassColorAttachment {
   738→                    view: &frame_view,
   739→                    resolve_target: None,
   740→                    ops: wgpu::Operations {
   741→                        load: wgpu::LoadOp::Clear(wgpu::Color::BLACK),
   742→                        store: wgpu::StoreOp::Store,
   743→                    },
   744→                })],
   745→                depth_stencil_attachment: None,
   746→                timestamp_writes: None,
   747→                occlusion_query_set: None,
   748→            });
   749→            pass.set_pipeline(&self.post_pipeline);
   750→            pass.set_bind_group(0, &self.post_bind, &[]);
   751→            pass.draw(0..3, 0..1);
   752→        }
   753→
   754→        // 3) UI -> surface.
   755→        if !verts.is_empty() {
   756→            let mut pass = encoder.begin_render_pass(&wgpu::RenderPassDescriptor {
   757→                label: Some("ui-pass"),
   758→                color_attachments: &[Some(wgpu::RenderPassColorAttachment {
   759→                    view: &frame_view,
   760→                    resolve_target: None,
   761→                    ops: wgpu::Operations {
   762→                        load: wgpu::LoadOp::Load,
   763→                        store: wgpu::StoreOp::Store,
   764→                    },
   765→                })],
   766→                depth_stencil_attachment: None,
   767→                timestamp_writes: None,
   768→                occlusion_query_set: None,
   769→            });
   770→            pass.set_pipeline(&self.ui_pipeline);
   771→            pass.set_bind_group(0, &self.ui_bind0, &[]);
   772→            let stride = std::mem::size_of::<ui::UiVertex>() as u64;
   773→            let mut cursor = 0u64;
   774→            for (_op, (is_text, quads)) in ui_ops.iter().zip(draws.iter()) {
   775→                if *quads == 0 {
   776→                    continue;
   777→                }
   778→                match is_text {
   779→                    true => pass.set_bind_group(1, &self.font_bg, &[]),
   780→                    false => pass.set_bind_group(1, &self.white_bg, &[]),
   781→                }
   782→                let bytes = *quads as u64 * 4 * stride;
   783→                pass.set_vertex_buffer(0, self.ui_buf.slice(cursor..cursor + bytes));
   784→                pass.draw(0..4, 0..*quads);
   785→                cursor += bytes;
   786→            }
   787→        }
   788→
   789→        self.queue.submit(Some(encoder.finish()));
   790→        frame.present();
   791→        Ok(())
   792→    }
   793→}
