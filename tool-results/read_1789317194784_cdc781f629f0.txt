    62→    62→impl Clone for AutoAct {
    63→    63→    fn clone(&self) -> Self {
    64→    64→        match self {
    65→    65→            AutoAct::Key(k) => AutoAct::Key(*k),
    66→    66→            AutoAct::Shot(p) => AutoAct::Shot(p.clone()),
    67→    67→            AutoAct::Yaw(a) => AutoAct::Yaw(*a),
    68→    68→            AutoAct::Exit => AutoAct::Exit,
    69→    69→        }
    70→    70→    }

    75→    75→    steps: Vec<AutoStep>,

    79→    79→fn parse_key(name: &str) -> Option<KeyCode> {
    80→    80→    Some(match name {
    81→    81→        "Enter" | "NumpadEnter" => KeyCode::Enter,
    82→    82→        "Digit1" => KeyCode::Digit1,
    83→    83→        "Digit2" => KeyCode::Digit2,
    84→    84→        "Digit3" => KeyCode::Digit3,
    85→    85→        "Space" => KeyCode::Space,
    86→    86→        "Escape" => KeyCode::Escape,
    87→    87→        "Backspace" => KeyCode::Backspace,
    88→    88→        "F5" => KeyCode::F5,
    89→    89→        "KeyF" => KeyCode::KeyF,
    90→    90→        "KeyE" => KeyCode::KeyE,
    91→    91→        _ => return None,
    92→    92→    })
    93→    93→}

    95→    95→/// Format : « key Digit1@1.0 ; shot /tmp/x.png@3.0 ; exit@3.2 »
    96→    96→fn parse_autopilot(s: &str) -> Option<Auto> {
    97→    97→    let mut steps = Vec::new();
    98→    98→    for tok in s.split(';') {
    99→    99→        let tok = tok.trim();
   100→   100→        if tok.is_empty() {
   101→   101→            continue;
   102→   102→        }
   103→   103→        let (head, at) = tok.rsplit_once('@')?;
   104→   104→        let at: f64 = at.trim().parse().ok()?;
   105→   105→        let mut parts = head.splitn(2, ' ');
   106→   106→        let verb = parts.next().unwrap_or("").trim();
   107→   107→        let arg = parts.next().unwrap_or("").trim();
   108→   108→        let act = match verb {
   109→   109→            "key" => AutoAct::Key(parse_key(arg)?),
   110→   110→            "shot" => AutoAct::Shot(arg.to_string()),
   111→   111→            "yaw" => AutoAct::Yaw(arg.parse::<f32>().ok()?.to_radians()),
   112→   112→            "exit" => AutoAct::Exit,
   113→   113→            _ => return None,
   114→   114→        };
   115→   115→        steps.push(AutoStep { at, act });
   116→   116→    }
   117→   117→    steps.sort_by(|a, b| a.at.partial_cmp(&b.at).unwrap_or(std::cmp::Ordering::Equal));
   118→   118→    if steps.is_empty() {
   119→   119→        None
   120→   120→    } else {
   121→   121→        Some(Auto { t0: Instant::now(), steps, next: 0 })
   122→   122→    }
   123→   123→}

   190→   190→    /// Cycle le mode ray tracing (off -> qualité -> ultra), applique et persiste.
   191→   191→    fn cycle_rt_mode(&mut self) -> &'static str {
   192→   192→        self.config.rt_mode = (self.config.rt_mode + 1) % 3;
   193→   193→        self.config.save();
   194→   194→        if let Some(r) = self.renderer.as_mut() {
   195→   195→            r.set_rt_mode(self.config.rt_mode);
   196→   196→        }
   197→   197→        match self.config.rt_mode {
   198→   198→            1 => t(self.lang, RT_QUAL),
   199→   199→            2 => t(self.lang, RT_ULTRA),
   200→   200→            _ => t(self.lang, RT_OFF),
   201→   201→        }
   202→   202→    }

   255→   255→    fn handle_net_events(&mut self) {
   256→   256→        let events: Vec<NetEvent> = match &self.net {

   269→   269→                }
   270→   270→                NetEvent::Msg(msg) => self.handle_msg(msg),
   271→   271→            }
   272→   272→        }
   273→   273→    }

   275→   275→    fn handle_msg(&mut self, msg: ServerMsg) {
   276→   276→        match msg {
   277→   277→            ServerMsg::Welcome { .. } => {
   278→   278→                if let Some(p) = self.pending.take() {
   279→   279→                    if let Some(n) = self.net.as_ref() {
   280→   280→                        match p {
   281→   281→                            Pending::Create { password } => n.send(ClientMsg::CreateRoom { password }),
   282→   282→                            Pending::Join { code, password } => {
   283→   283→                                n.send(ClientMsg::JoinRoom { code, password })
   284→   284→                            }
   285→   285→                        }
   286→   286→                    }
   287→   287→                }
   288→   288→            }
   289→   289→            ServerMsg::RoomCreated { .. } => {}
   290→   290→            ServerMsg::RoomJoined { code, players, is_host } => {
   291→   291→                self.mode = Mode::Lobby { code, is_host };
   292→   292→                self.players = players;
   293→   293→            }
   294→   294→            ServerMsg::LobbyUpdate { players } => self.players = players,
   295→   295→            ServerMsg::RoomError { msg_key } => {
   296→   296→                let key = match msg_key {
   297→   297→                    ERR_ROOM_NOT_FOUND => E_ERR_NOT_FOUND,
   298→   298→                    ERR_WRONG_PASSWORD => E_ERR_PASSWORD,
   299→   299→                    ERR_ROOM_FULL => E_ERR_FULL,
   300→   300→                    ERR_GAME_RUNNING => E_ERR_GAME,
   301→   301→                    ERR_NO_ROOM => E_ERR_NO_ROOM,
   302→   302→                    ERR_NOT_HOST => E_ERR_NOT_HOST,
   303→   303→                    _ => E_ERR_CLOSED,
   304→   304→                };
   305→   305→                self.show_err(t(self.lang, key).to_string());
   306→   306→                self.mode = Mode::MainMenu(MenuScreen::Main);
   307→   307→            }
   308→   308→            ServerMsg::GameStarted { your_id, spawn, snapshot } => {
   309→   309→                let map: &'static MapData = Box::leak(Box::new(MapData::parse()));
   310→   310→                self.map = Some(map);
   311→   311→                if let Some(r) = self.renderer.as_mut() {
   312→   312→                    let mut g = Game::new(map, your_id, spawn, r);
   313→   313→                    g.apply_snapshot(snapshot);
   314→   314→                    self.game = Some(Box::new(g));
   315→   315→                    self.mode = Mode::Playing;
   316→   316→                    self.paused = false;
   317→   317→                    self.set_cursor_locked(true);
   318→   318→                    if let Some(a) = self.audio.as_mut() {
   319→   319→                        a.start_loop("ambience", 0.5);
   320→   320→                        a.start_loop("fluorescent", 0.12);
   321→   321→                    }
   322→   322→                }
   323→   323→            }
   324→   324→            ServerMsg::Snapshot(snap) => {
   325→   325→                if let Some(g) = self.game.as_mut() {
   326→   326→                    g.apply_snapshot(snap);
   327→   327→                }
   328→   328→            }
   329→   329→            ServerMsg::Announcement { fr, en } => {
   330→   330→                if let Some(g) = self.game.as_mut() {
   331→   331→                    let msg = match self.lang {
   332→   332→                        Lang::Fr => fr,
   333→   333→                        Lang::En => en,
   334→   334→                    };
   335→   335→                    g.add_feed(msg);
   336→   336→                }
   337→   337→            }
   338→   338→            ServerMsg::Sfx { kind, pos } => self.play_sfx(kind, pos),
   339→   339→            ServerMsg::Event(_) => {}
   340→   340→            ServerMsg::GameOver { win, time, servers_fixed, escaped } => {
   341→   341→                self.mode = Mode::GameOver { win, time, servers: servers_fixed, escaped };
   342→   342→                self.set_cursor_locked(false);
   343→   343→                if let Some(a) = self.audio.as_mut() {
   344→   344→                    a.stop_loop("ambience");
   345→   345→                    a.stop_loop("fluorescent");
   346→   346→                    a.stop_loop("whisper");
   347→   347→                }
   348→   348→            }
   349→   349→            ServerMsg::Pong { .. } => {}
   350→   350→        }
   351→   351→    }

   353→   353→    fn play_sfx(&mut self, kind: SfxKind, pos: Vec3) {
   354→   354→        let my = self.game.as_ref().map(|g| g.pos);
   355→   355→        let gain = match my {
   356→   356→            Some(my) => (1.0 - my.distance(pos) / 26.0).clamp(0.0, 1.0),
   357→   357→            None => 0.5,
   358→   358→        };
   359→   359→        if let Some(a) = &self.audio {
   360→   360→            let name = match kind {
   361→   361→                SfxKind::EntityStep => {
   362→   362→                    if rand_int(2) == 0 {
   363→   363→                        "entity_step1"
   364→   364→                    } else {
   365→   365→                        "entity_step2"
   366→   366→                    }
   367→   367→                }
   368→   368→                SfxKind::DoorOpen => "door_open",
   369→   369→                SfxKind::DoorSlam => "door_slam",
   370→   370→                SfxKind::Jumpscare => "jumpscare",
   371→   371→                SfxKind::Screech => "screech",
   372→   372→                SfxKind::RebootDone => "reboot_done",
   373→   373→                SfxKind::Pickup => "pickup",
   374→   374→                SfxKind::Stamp => "stamp",
   375→   375→                SfxKind::Battery => "battery",
   376→   376→                SfxKind::Blackout => "blackout",
   377→   377→                SfxKind::LightsOn => "lights_on",
   378→   378→                SfxKind::Breaker => "breaker",
   379→   379→                SfxKind::Caught => "caught",
   380→   380→                SfxKind::Whisper => "whisper",
   381→   381→            };
   382→   382→            let g2 = match kind {
   383→   383→                SfxKind::Jumpscare | SfxKind::Caught => 1.0,
   384→   384→                SfxKind::Whisper => gain * 0.5,
   385→   385→                _ => gain,
   386→   386→            };
   387→   387→            a.play(name, g2);
   388→   388→        }
   389→   389→    }

   393→   393→    fn on_key_pressed(&mut self, code: KeyCode, _el: &ActiveEventLoop) {
   394→   394→        if code == KeyCode::F12 {
   395→   395→            // Capture d'écran instantanée -> PNG à côté de l'exécutable.
   396→   396→            let ms = std::time::SystemTime::now()
   397→   397→                .duration_since(std::time::UNIX_EPOCH)
   398→   398→                .map(|d| d.as_millis())
   399→   399→                .unwrap_or(0);
   400→   400→            if let Some(r) = &mut self.renderer {
   401→   401→                r.capture_path = Some(std::path::PathBuf::from(format!("sl3_shot_{ms}.png")));
   402→   402→            }
   403→   403→            return;
   404→   404→        }
   405→   405→        if code == KeyCode::F1 {
   406→   406→            self.lang.toggle();
   407→   407→            self.config.lang = match self.lang {
   408→   408→                Lang::Fr => "fr".into(),
   409→   409→                Lang::En => "en".into(),
   410→   410→            };
   411→   411→            self.config.save();
   412→   412→            return;
   413→   413→        }
   414→   414→
   415→   415→        let in_menu = matches!(self.mode, Mode::MainMenu(_));
   416→   416→        let in_lobby = matches!(self.mode, Mode::Lobby { .. });
   417→   417→        let in_game = matches!(self.mode, Mode::Playing);
   418→   418→        let in_over = matches!(self.mode, Mode::GameOver { .. });
   419→   419→
   420→   420→        if in_menu {
   421→   421→            if let Mode::MainMenu(screen) = &self.mode {
   422→   422→                let screen = screen.clone();
   423→   423→                self.menu_key(&screen, code);
   424→   424→            }
   425→   425→        } else if in_lobby {
   426→   426→            match code {
   427→   427→                KeyCode::Space => {
   428→   428→                    if let Some(n) = &self.net {
   429→   429→                        n.send(ClientMsg::StartGame);
   430→   430→                    }
   431→   431→                }
   432→   432→                KeyCode::Escape => self.leave_room(),
   433→   433→                _ => {}
   434→   434→            }
   435→   435→        } else if in_game {
   436→   436→            match code {
   437→   437→                KeyCode::Escape => {
   438→   438→                    self.paused = !self.paused;
   439→   439→                    self.set_cursor_locked(!self.paused);
   440→   440→                }
   441→   441→                KeyCode::KeyF if !self.paused => {
   442→   442→                    if let Some(g) = self.game.as_mut() {
   443→   443→                        if g.battery > 0.0 {
   444→   444→                            g.light_on = !g.light_on;
   445→   445→                        }
   446→   446→                    }
   447→   447→                }
   448→   448→                KeyCode::KeyE if !self.paused => {
   449→   449→                    self.interact_pressed();
   450→   450→                }
   451→   451→                KeyCode::F5 => {
   452→   452→                    let label = self.cycle_rt_mode();
   453→   453→                    if let Some(g) = self.game.as_mut() {
   454→   454→                        g.add_feed(format!("{} : {}", t(self.lang, RT_TOAST), label));
   455→   455→                    }
   456→   456→                }
   457→   457→                _ => {}
   458→   458→            }
   459→   459→        } else if in_over && code == KeyCode::Enter {
   460→   460→            self.leave_room();
   461→   461→        }
   462→   462→    }

   484→   484→    fn menu_key(&mut self, screen: &MenuScreen, code: KeyCode) {
   485→   485→        let enter = code == KeyCode::Enter || code == KeyCode::NumpadEnter;
   486→   486→        let back = code == KeyCode::Escape;
   487→   487→        match screen {
   488→   488→            MenuScreen::Main => match code {
   489→   489→                KeyCode::Digit1 => {
   490→   490→                    self.mode = Mode::MainMenu(MenuScreen::AskHostAddr {
   491→   491→                        buf: self.config.host_default.clone(),
   492→   492→                    });
   493→   493→                }
   494→   494→                KeyCode::Digit2 => {
   495→   495→                    self.mode = Mode::MainMenu(MenuScreen::AskJoinAddr {
   496→   496→                        buf: self.config.host_default.clone(),
   497→   497→                    });
   498→   498→                }
   499→   499→                KeyCode::Digit3 => {
   500→   500→                    self.mode = Mode::MainMenu(MenuScreen::Options);
   501→   501→                }
   502→   502→                KeyCode::Escape => std::process::exit(0),
   503→   503→                _ => {}
   504→   504→            },
   505→   505→            MenuScreen::AskHostAddr { buf } => {
   506→   506→                if back {
   507→   507→                    self.back_to_menu();
   508→   508→                } else if code == KeyCode::Backspace {
   509→   509→                    let mut b = buf.clone();
   510→   510→                    b.pop();
   511→   511→                    self.mode = Mode::MainMenu(MenuScreen::AskHostAddr { buf: b });
   512→   512→                } else if enter {
   513→   513→                    self.config.host_default = buf.clone();
   514→   514→                    self.config.save();
   515→   515→                    self.mode = Mode::MainMenu(MenuScreen::HostPassword { buf: String::new() });
   516→   516→                }
   517→   517→            }
   518→   518→            MenuScreen::HostPassword { buf } => {
   519→   519→                if back {
   520→   520→                    self.back_to_menu();
   521→   521→                } else if code == KeyCode::Backspace {
   522→   522→                    let mut b = buf.clone();
   523→   523→                    b.pop();
   524→   524→                    self.mode = Mode::MainMenu(MenuScreen::HostPassword { buf: b });
   525→   525→                } else if enter {
   526→   526→                    let password = buf.clone();
   527→   527→                    let addr = self.config.host_default.clone();
   528→   528→                    self.connect(addr, Pending::Create { password });
   529→   529→                }
   530→   530→            }
   531→   531→            MenuScreen::AskJoinAddr { buf } => {
   532→   532→                if back {
   533→   533→                    self.back_to_menu();
   534→   534→                } else if code == KeyCode::Backspace {
   535→   535→                    let mut b = buf.clone();
   536→   536→                    b.pop();
   537→   537→                    self.mode = Mode::MainMenu(MenuScreen::AskJoinAddr { buf: b });
   538→   538→                } else if enter {
   539→   539→                    self.config.host_default = buf.clone();
   540→   540→                    self.config.save();
   541→   541→                    self.mode = Mode::MainMenu(MenuScreen::AskCode { buf: String::new() });
   542→   542→                }
   543→   543→            }
   544→   544→            MenuScreen::AskCode { buf } => {
   545→   545→                if back {
   546→   546→                    self.back_to_menu();
   547→   547→                } else if code == KeyCode::Backspace {
   548→   548→                    let mut b = buf.clone();
   549→   549→                    b.pop();
   550→   550→                    self.mode = Mode::MainMenu(MenuScreen::AskCode { buf: b });
   551→   551→                } else if enter {
   552→   552→                    let code_s = buf.trim().to_uppercase();
   553→   553→                    self.mode = Mode::MainMenu(MenuScreen::AskPassword {
   554→   554→                        code: code_s,
   555→   555→                        buf: String::new(),
   556→   556→                    });
   557→   557→                }
   558→   558→            }
   559→   559→            MenuScreen::AskPassword { code: room_code, buf } => {
   560→   560→                if back {
   561→   561→                    self.back_to_menu();
   562→   562→                } else if code == KeyCode::Backspace {
   563→   563→                    let mut b = buf.clone();
   564→   564→                    b.pop();
   565→   565→                    self.mode = Mode::MainMenu(MenuScreen::AskPassword {
   566→   566→                        code: room_code.clone(),
   567→   567→                        buf: b,
   568→   568→                    });
   569→   569→                } else if enter {
   570→   570→                    let password = buf.clone();
   571→   571→                    let code_s = room_code.clone();
   572→   572→                    let addr = self.config.host_default.clone();
   573→   573→                    self.connect(addr, Pending::Join { code: code_s, password });
   574→   574→                }
   575→   575→            }
   576→   576→            MenuScreen::AskName { buf } => {
   577→   577→                if back || enter {
   578→   578→                    if enter && !buf.trim().is_empty() {
   579→   579→                        self.config.name = buf.trim().chars().take(MAX_NAME_LEN).collect();
   580→   580→                        self.config.save();
   581→   581→                    }
   582→   582→                    self.mode = Mode::MainMenu(MenuScreen::Options);
   583→   583→                } else if code == KeyCode::Backspace {
   584→   584→                    let mut b = buf.clone();
   585→   585→                    b.pop();
   586→   586→                    self.mode = Mode::MainMenu(MenuScreen::AskName { buf: b });
   587→   587→                }
   588→   588→            }
   589→   589→            MenuScreen::Options => match code {
   590→   590→                KeyCode::Escape | KeyCode::Enter => {
   591→   591→                    self.mode = Mode::MainMenu(MenuScreen::Main);
   592→   592→                }
   593→   593→                KeyCode::KeyL => {
   594→   594→                    self.lang.toggle();
   595→   595→                    self.config.lang = match self.lang {
   596→   596→                        Lang::Fr => "fr".into(),
   597→   597→                        Lang::En => "en".into(),
   598→   598→                    };
   599→   599→                    self.config.save();
   600→   600→                }
   601→   601→                KeyCode::KeyN => {
   602→   602→                    self.mode = Mode::MainMenu(MenuScreen::AskName {
   603→   603→                        buf: self.config.name.clone(),
   604→   604→                    });
   605→   605→                }
   606→   606→                KeyCode::KeyR => {
   607→   607→                    // Cycle : auto -> 100 % -> 85 % -> 70 % -> 55 % -> auto.
   608→   608→                    self.config.render_scale = if self.config.render_scale <= 0.0 {
   609→   609→                        1.0
   610→   610→                    } else if (self.config.render_scale - 1.0).abs() < 1e-3 {
   611→   611→                        0.85
   612→   612→                    } else if (self.config.render_scale - 0.85).abs() < 1e-3 {
   613→   613→                        0.7
   614→   614→                    } else if (self.config.render_scale - 0.7).abs() < 1e-3 {
   615→   615→                        0.55
   616→   616→                    } else {
   617→   617→                        0.0
   618→   618→                    };
   619→   619→                    self.config.save();
   620→   620→                    if self.config.render_scale > 0.0 {
   621→   621→                        if let Some(r) = self.renderer.as_mut() {
   622→   622→                            r.set_render_scale(self.config.render_scale);
   623→   623→                        }
   624→   624→                    }
   625→   625→                }
   626→   626→                KeyCode::KeyT => {
   627→   627→                    let _ = self.cycle_rt_mode();
   628→   628→                }
   629→   629→                KeyCode::ArrowLeft => {
   630→   630→                    self.config.sensitivity = (self.config.sensitivity - 0.1).max(0.2);
   631→   631→                    self.config.save();
   632→   632→                }
   633→   633→                KeyCode::ArrowRight => {
   634→   634→                    self.config.sensitivity = (self.config.sensitivity + 0.1).min(3.0);
   635→   635→                    self.config.save();
   636→   636→                }
   637→   637→                _ => {}
   638→   638→            },
   639→   639→        }
   640→   640→    }

   643→   643→
   644→   644→    fn frame(&mut self) {
   645→   645→        if self.renderer.is_none() {
   646→   646→            return;
   647→   647→        }
   648→   648→        self.handle_net_events();
   649→   649→        let dt = self.last_frame.elapsed().as_secs_f32().min(0.1);
   650→   650→        self.last_frame = Instant::now();
   651→   651→
   652→   652→        // Résolution dynamique : adapte l'échelle de rendu au temps de frame réel
   653→   653→        // (cible ~60 fps ; plafond 1.0, plancher 0.45) — pensé pour iGPU / vieilles machines.
   654→   654→        self.ema_frame = self.ema_frame * 0.92 + dt * 0.08;
   655→   655→        if let Some(r) = self.renderer.as_mut() {
   656→   656→            if self.config.render_scale <= 0.0 {
   657→   657→                if self.last_scale_adj.elapsed().as_secs_f32() > 0.75 {
   658→   658→                    let cur = r.render_scale();
   659→   659→                    if self.ema_frame > 0.021 && cur > 0.45 {
   660→   660→                        r.set_render_scale(cur - 0.05);
   661→   661→                        self.last_scale_adj = Instant::now();
   662→   662→                    } else if self.ema_frame < 0.0145 && cur < 1.0 {
   663→   663→                        r.set_render_scale(cur + 0.05);
   664→   664→                        self.last_scale_adj = Instant::now();
   665→   665→                    }
   666→   666→                }
   667→   667→            } else {
   668→   668→                r.set_render_scale(self.config.render_scale);
   669→   669→            }
   670→   670→        }
   671→   671→        let (w, h) = {
   672→   672→            let s = self.renderer.as_ref().unwrap().size();
   673→   673→            (s.0 as f32, s.1 as f32)
   674→   674→        };
   675→   675→        let font = self.renderer.as_ref().unwrap().font.clone();
   676→   676→        let mut ui_ops: Vec<UiOp> = Vec::new();
   677→   677→
   678→   678→        let tag = match &self.mode {
   679→   679→            Mode::MainMenu(_) => 0,
   680→   680→            Mode::Lobby { .. } => 1,
   681→   681→            Mode::Playing => 2,
   682→   682→            Mode::GameOver { .. } => 3,
   683→   683→        };
   684→   684→
   685→   685→        let mut world_data: Option<(
   686→   686→            crate::gpu::WorldUniform,
   687→   687→            [f32; 4],
   688→   688→            Vec<(String, Vec<crate::gpu::InstanceData>)>,
   689→   689→            [[f32; 4]; 4],
   690→   690→            Vec<crate::gpu::rtscene::GpuAabb>,
   691→   691→        )> = None;
   692→   692→
   693→   693→        match tag {
   694→   694→            0 => {
   695→   695→                let screen = match &self.mode {
   696→   696→                    Mode::MainMenu(s) => s.clone(),
   697→   697→                    _ => MenuScreen::Main,
   698→   698→                };
   699→   699→                self.draw_menu(&mut ui_ops, &font, &screen, (w, h));
   700→   700→            }
   701→   701→            1 => {
   702→   702→                let (code, is_host) = match &self.mode {
   703→   703→                    Mode::Lobby { code, is_host } => (code.clone(), *is_host),
   704→   704→                    _ => (String::new(), false),
   705→   705→                };
   706→   706→                self.draw_lobby(&mut ui_ops, &font, &code, is_host, (w, h));
   707→   707→            }
   708→   708→            2 => {
   709→   709→                let Some(tx) = self.net.as_ref().map(|n| n.tx.clone()) else { return };
   710→   710→                let (wu, post, dyns, fear, inv_vp, rt_boxes) = {
   711→   711→                    let Some(g) = self.game.as_mut() else { return };
   712→   712→                    let local_sfx = g.update(dt, &self.keys, &tx);
   713→   713→                    if let Some(a) = self.audio.as_ref() {
   714→   714→                        for (name, gain) in local_sfx {
   715→   715→                            a.play(&name, gain);
   716→   716→                        }
   717→   717→                    }
   718→   718→                    let dyns = g.dynamics();
   719→   719→                    let wu = g.world_uniform(w / h);
   720→   720→                    let inv_vp = g.inv_view_proj(w / h);
   721→   721→                    let rt_boxes = match self.renderer.as_ref() {
   722→   722→                        Some(r) => g.rt_dynamic_boxes(&r.models),
   723→   723→                        None => Vec::new(),
   724→   724→                    };
   725→   725→                    let post = g.post_params();
   726→   726→                    (wu, post, dyns, g.fear, inv_vp, rt_boxes)
   727→   727→                };
   728→   728→                if let Some(a) = self.audio.as_ref() {
   729→   729→                    a.set_loop_volume("whisper", fear * 0.6);
   730→   730→                }
   731→   731→                if let Some(g) = self.game.as_ref() {
   732→   732→                    hud::draw(&mut ui_ops, g, &font, self.lang, (w, h));
   733→   733→                }
   734→   734→                // Ligne perf (haut-droite) : FPS + échelle de rendu + indicateur RT.
   735→   735→                let fps = (1.0 / self.ema_frame.max(1e-4)) as u32;
   736→   736→                let pct = (self.renderer.as_ref().unwrap().render_scale() * 100.0).round() as u32;
   737→   737→                let rt_tag = if self.config.rt_mode > 0 { " – RT" } else { "" };
   738→   738→                let perf = format!("{fps} FPS – rendu {pct}%{rt_tag}");
   739→   739→                let tw = crate::gpu::ui::text_width(&font, &perf, 13.0);
   740→   740→                ui_ops.push(UiOp::text(w - tw - 12.0, 10.0, 13.0, [0.62, 0.68, 0.62, 0.75], &perf));
   741→   741→                if self.paused {
   742→   742→                    draw_center(&mut ui_ops, &font, t(self.lang, PAUSED), 34.0, WHITE_C, (w, h * 0.42));
   743→   743→                    draw_center(&mut ui_ops, &font, t(self.lang, RESUME_HINT), 18.0, DIM_C, (w, h * 0.42 + 50.0));
   744→   744→                    draw_center(&mut ui_ops, &font, t(self.lang, QUIT_HINT), 18.0, DIM_C, (w, h * 0.42 + 78.0));
   745→   745→                }
   746→   746→                world_data = Some((wu, post, dyns, inv_vp, rt_boxes));
   747→   747→            }
   748→   748→            _ => {
   749→   749→                let (win, time, servers, escaped) = match &self.mode {
   750→   750→                    Mode::GameOver { win, time, servers, escaped } => (*win, *time, *servers, *escaped),
   751→   751→                    _ => (false, 0.0, 0, 0),
   752→   752→                };
   753→   753→                self.draw_gameover(&mut ui_ops, &font, win, time, servers, escaped, (w, h));
   754→   754→            }
   755→   755→        }
   756→   756→
   757→   757→        // Erreur temporaire.
   758→   758→        if let Some((at, msg)) = &self.err {
   759→   759→            let age = at.elapsed().as_secs_f32();
   760→   760→            if age < 4.0 {
   761→   761→                let alpha = 1.0 - age / 4.0;
   762→   762→                draw_center(&mut ui_ops, &font, msg, 18.0, [1.0, 0.45, 0.4, alpha], (w, h * 0.82));
   763→   763→            } else {
   764→   764→                self.err = None;
   765→   765→            }
   766→   766→        }
   767→   767→
   768→   768→        // Rendu final.
   769→   769→        match world_data {
   770→   770→            Some((wu, post, dyns, inv_vp, rt_boxes)) => {
   771→   771→                let statics: &crate::gpu::StaticBatches = match self.game.as_ref() {
   772→   772→                    Some(g) => &g.statics,
   773→   773→                    None => empty_statics(),
   774→   774→                };
   775→   775→                let renderer = self.renderer.as_mut().unwrap();
   776→   776→                let rt = if renderer.rt_mode > 0 {
   777→   777→                    Some(crate::gpu::RtFrame {
   778→   778→                        inv_vp,
   779→   779→                        dyn_boxes: &rt_boxes,
   780→   780→                    })
   781→   781→                } else {
   782→   782→                    None
   783→   783→                };
   784→   784→                let _ = renderer.render(&wu, rt, statics, &dyns, &ui_ops, post);
   785→   785→            }
   786→   786→            None => {
   787→   787→                let _ = self.renderer.as_mut().unwrap().render(
   788→   788→                    &identity_uniform(),
   789→   789→                    None,
   790→   790→                    empty_statics(),
   791→   791→                    &Vec::new(),
   792→   792→                    &ui_ops,
   793→   793→                    [0.0, 0.0, 0.0, 0.0],
   794→   794→                );
   795→   795→            }
   796→   796→        }
   797→   797→    }

  1002→  1002→fn rand_int(n: u32) -> u32 {
  1003→  1003→    use std::cell::Cell;
  1004→  1004→    thread_local! {
  1005→  1005→        static S: Cell<u32> = Cell::new(0x9e3779b9);
  1006→  1006→    }
  1007→  1007→    S.with(|s| {
  1008→  1008→        let mut x = s.get();
  1009→  1009→        x ^= x << 13;
  1010→  1010→        x ^= x >> 17;
  1011→  1011→        x ^= x << 5;
  1012→  1012→        s.set(x);
  1013→  1013→        x % n.max(1)
  1014→  1014→    })
  1015→  1015→}

  1017→  1017→impl App {
  1018→  1018→    fn pump_autopilot(&mut self, el: &ActiveEventLoop) {
  1019→  1019→        let mut due: Vec<AutoAct> = Vec::new();
  1020→  1020→        if let Some(auto) = self.auto.as_mut() {
  1021→  1021→            let t = auto.t0.elapsed().as_secs_f64();
  1022→  1022→            while auto.next < auto.steps.len() && auto.steps[auto.next].at <= t {
  1023→  1023→                due.push(auto.steps[auto.next].act.clone());
  1024→  1024→                auto.next += 1;
  1025→  1025→            }
  1026→  1026→        } else {
  1027→  1027→            return;
  1028→  1028→        }
  1029→  1029→        for act in due {
  1030→  1030→            match act {
  1031→  1031→                AutoAct::Key(c) => {
  1032→  1032→                    self.keys.insert(c);
  1033→  1033→                    self.on_key_pressed(c, el);
  1034→  1034→                    self.keys.remove(&c);
  1035→  1035→                    self.on_key_released(c);
  1036→  1036→                }
  1037→  1037→                AutoAct::Shot(p) => {
  1038→  1038→                    if let Some(r) = self.renderer.as_mut() {
  1039→  1039→                        r.capture_path = Some(std::path::PathBuf::from(&p));
  1040→  1040→                        println!("[autopilot] capture -> {p}");
  1041→  1041→                    }
  1042→  1042→                }
  1043→  1043→                AutoAct::Yaw(a) => {
  1044→  1044→                    if let Some(g) = self.game.as_mut() {
  1045→  1045→                        g.yaw += a;
  1046→  1046→                    }
  1047→  1047→                }
  1048→  1048→                AutoAct::Exit => {
  1049→  1049→                    self.config.save();
  1050→  1050→                    println!("[autopilot] fin");
  1051→  1051→                    el.exit();
  1052→  1052→                }
  1053→  1053→            }
  1054→  1054→        }
  1055→  1055→    }
  1056→  1056→}

  1079→  1079→    fn window_event(&mut self, el: &ActiveEventLoop, _id: winit::window::WindowId, event: WindowEvent) {
  1080→  1080→        match event {
  1081→  1081→            WindowEvent::CloseRequested => {
  1082→  1082→                self.config.save();
  1083→  1083→                el.exit();
  1084→  1084→            }
  1085→  1085→            WindowEvent::Resized(size) => {
  1086→  1086→                if let Some(r) = &mut self.renderer {
  1087→  1087→                    r.resize(size.width, size.height);
  1088→  1088→                }
  1089→  1089→            }
  1090→  1090→            WindowEvent::CursorMoved { position, .. } => {
  1091→  1091→                if self.cursor_locked {
  1092→  1092→                    if let (Some(win), Some(r)) = (&self.window, &self.renderer) {
  1093→  1093→                        let (w, h) = r.size();
  1094→  1094→                        let cx = w as f64 / 2.0;
  1095→  1095→                        let cy = h as f64 / 2.0;
  1096→  1096→                        let dx = position.x - cx;
  1097→  1097→                        let dy = position.y - cy;
  1098→  1098→                        if dx.abs() < 600.0 && dy.abs() < 600.0 {
  1099→  1099→                            if let Some(g) = self.game.as_mut() {
  1100→  1100→                                let sens = self.config.sensitivity;
  1101→  1101→                                g.yaw += (dx as f32) * 0.0022 * sens;
  1102→  1102→                                g.pitch -= (dy as f32) * 0.0022 * sens;
  1103→  1103→                                g.pitch = g.pitch.clamp(-1.45, 1.45);
  1104→  1104→                            }
  1105→  1105→                        }
  1106→  1106→                        if let Ok(op) = win.outer_position() {
  1107→  1107→                            let _ = win.set_cursor_position(PhysicalPosition::new(
  1108→  1108→                                op.x + cx as i32,
  1109→  1109→                                op.y + cy as i32,
  1110→  1110→                            ));
  1111→  1111→                        }
  1112→  1112→                    }
  1113→  1113→                }
  1114→  1114→            }
  1115→  1115→            WindowEvent::KeyboardInput { event, .. } => {
  1116→  1116→                let code = match event.physical_key {
  1117→  1117→                    PhysicalKey::Code(c) => c,
  1118→  1118→                    _ => return,
  1119→  1119→                };
  1120→  1120→                match event.state {
  1121→  1121→                    ElementState::Pressed => {
  1122→  1122→                        if self.keys.insert(code) {
  1123→  1123→                            self.on_key_pressed(code, el);
  1124→  1124→                        }
  1125→  1125→                    }
  1126→  1126→                    ElementState::Released => {
  1127→  1127→                        self.keys.remove(&code);
  1128→  1128→                        self.on_key_released(code);
  1129→  1129→                    }
  1130→  1130→                }
  1131→  1131→            }
  1132→  1132→            WindowEvent::RedrawRequested => {
  1133→  1133→                self.frame();
  1134→  1134→            }
  1135→  1135→            _ => {}
  1136→  1136→        }
  1137→  1137→    }
