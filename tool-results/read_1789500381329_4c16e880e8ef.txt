     1→#!/bin/sh
     2→# shellcheck shell=dash
     3→# shellcheck disable=SC2039  # local is non-POSIX
     4→
     5→# This is just a little script that can be downloaded from the internet to
     6→# install rustup. It just does platform detection, downloads the installer
     7→# and runs it.
     8→
     9→# It runs on Unix shells like {a,ba,da,k,z}sh. It uses the common `local`
    10→# extension. Note: Most shells limit `local` to 1 var per line, contra bash.
    11→
    12→# Some versions of ksh have no `local` keyword. Alias it to `typeset`, but
    13→# beware this makes variables global with f()-style function syntax in ksh93.
    14→# mksh has this alias by default.
    15→has_local() {
    16→    # shellcheck disable=SC2034  # deliberately unused
    17→    local _has_local
    18→}
    19→
    20→has_local 2>/dev/null || alias local=typeset
    21→
    22→is_zsh() {
    23→    [ -n "${ZSH_VERSION-}" ]
    24→}
    25→
    26→set -u
    27→
    28→# If RUSTUP_UPDATE_ROOT is unset or empty, default it.
    29→RUSTUP_UPDATE_ROOT="${RUSTUP_UPDATE_ROOT:-https://static.rust-lang.org/rustup}"
    30→# Set quiet as a global for ease of use
    31→RUSTUP_QUIET=no
    32→
    33→# NOTICE: If you change anything here, please make the same changes in setup_mode.rs
    34→usage() {
    35→    cat <<EOF
    36→rustup-init 1.29.1 (3f5811cec 2026-08-12)
    37→
    38→The installer for rustup
    39→
    40→Usage: rustup-init[EXE] [OPTIONS]
    41→
    42→Options:
    43→  -v, --verbose
    44→          Set log level to 'DEBUG' if 'RUSTUP_LOG' is unset
    45→  -q, --quiet
    46→          Disable progress output, set log level to 'WARN' if 'RUSTUP_LOG' is unset
    47→  -y, --yes
    48→          Disable confirmation prompt
    49→      --default-host <DEFAULT_HOST>
    50→          Choose a default host tuple
    51→      --default-toolchain <DEFAULT_TOOLCHAIN>
    52→          Choose a default toolchain to install. Use 'none' to not install any toolchains at all
    53→      --profile <PROFILE>
    54→          [default: default] [possible values: minimal, default, complete]
    55→  -c, --component <COMPONENT>
    56→          Comma-separated list of component names to also install
    57→  -t, --target <TARGET>

    59→      --no-update-default-toolchain
    60→          Don't update any existing default toolchain after install
    61→      --no-modify-path

    63→  -h, --help

    65→  -V, --version
    66→          Print version
    67→EOF
    68→}

    70→main() {
    71→    downloader --check
    72→    need_cmd uname
    73→    need_cmd mktemp
    74→    need_cmd chmod
    75→    need_cmd mkdir
    76→    need_cmd rm
    77→    need_cmd rmdir
    78→
    79→    get_architecture || return 1
    80→    local _arch="$RETVAL"
    81→    assert_nz "$_arch" "arch"
    82→
    83→    local _default_host_override=""
    84→    case "$_arch" in
    85→        *windows*)
    86→            _default_host_override="--default-host=$_arch"
    87→            ;;
    88→    esac
    89→
    90→    local _ext=""
    91→    case "$_arch" in
    92→        *windows*)
    93→            _ext=".exe"
    94→            ;;
    95→    esac
    96→
    97→    local _url
    98→    if [ "${RUSTUP_VERSION+set}" = 'set' ]; then
    99→        say "\`RUSTUP_VERSION\` has been set to \`${RUSTUP_VERSION}\`"
   100→        _url="${RUSTUP_UPDATE_ROOT}/archive/${RUSTUP_VERSION}"
   101→    else
   102→        _url="${RUSTUP_UPDATE_ROOT}/dist"
   103→    fi
   104→    _url="${_url}/${_arch}/rustup-init${_ext}"
   105→
   106→
   107→    local _dir
   108→    if ! _dir="$(ensure mktemp -d)"; then
   109→        # Because the previous command ran in a subshell, we must manually
   110→        # propagate exit status.
   111→        exit 1
   112→    fi
   113→    local _file="${_dir}/rustup-init${_ext}"
   114→
   115→    local _ansi_escapes_are_valid=false
   116→    if [ -t 2 ]; then
   117→        if [ "${TERM+set}" = 'set' ]; then
   118→            case "$TERM" in
   119→                xterm*|rxvt*|urxvt*|linux*|vt*)
   120→                    _ansi_escapes_are_valid=true
   121→                ;;
   122→            esac
   123→        fi
   124→    fi
   125→
   126→    # check if we have to use /dev/tty to prompt the user
   127→    local need_tty=yes
   128→    for arg in "$@"; do
   129→        case "$arg" in
   130→            --help)
   131→                usage
   132→                exit 0
   133→                ;;
   134→            --quiet)
   135→                RUSTUP_QUIET=yes
   136→                ;;
   137→            --yes)
   138→                # user wants to skip the prompt --
   139→                # we don't need /dev/tty
   140→                need_tty=no
   141→                ;;
   142→            *)
   143→                OPTIND=1
   144→                if [ "${arg%%--*}" = "" ]; then
   145→                    # Long option (other than --help);
   146→                    # don't attempt to interpret it.
   147→                    continue
   148→                fi
   149→                while getopts :hqy sub_arg "$arg"; do
   150→                    case "$sub_arg" in
   151→                        h)
   152→                            usage
   153→                            exit 0
   154→                            ;;
   155→                        q)
   156→                            RUSTUP_QUIET=yes
   157→                            ;;
   158→                        y)
   159→                            # user wants to skip the prompt --
   160→                            # we don't need /dev/tty
   161→                            need_tty=no
   162→                            ;;
   163→                        *)
   164→                            ;;
   165→                        esac
   166→                done
   167→                ;;
   168→        esac
   169→    done
   170→
   171→    say 'downloading installer'
   172→
   173→    ensure mkdir -p "$_dir"
   174→    ensure downloader "$_url" "$_file" "$_arch"
   175→    ensure chmod u+x "$_file"
   176→    if [ ! -x "$_file" ]; then
   177→        err "Cannot execute $_file (likely because of mounting /tmp as noexec)."
   178→        err "Please copy the file to a location where you can execute binaries and run ./rustup-init${_ext}."
   179→        exit 1
   180→    fi
   181→
   182→    # If a default host has been specified on the command line, we should
   183→    # revert our own override before calling the installer.
   184→    for arg in "$@"; do
   185→        case "$arg" in
   186→            --default-host|--default-host=*)
   187→                _default_host_override=
   188→                break
   189→                ;;
   190→        esac
   191→    done
   192→
   193→    if [ "$need_tty" = "yes" ] && [ ! -t 0 ]; then
   194→        # The installer is going to want to ask for confirmation by
   195→        # reading stdin.  This script was piped into `sh` though and
   196→        # doesn't have stdin to pass to its children. Instead we're going
   197→        # to explicitly connect /dev/tty to the installer's stdin.
   198→        if [ ! -t 1 ]; then
   199→            err "Unable to run interactively. Run with -y to accept defaults, --help for additional options"
   200→            exit 1;
   201→        fi
   202→
   203→        ignore "$_file" ${_default_host_override:+"$_default_host_override"} "$@" < /dev/tty
   204→    else
   205→        ignore "$_file" ${_default_host_override:+"$_default_host_override"} "$@"
   206→    fi
   207→
   208→    local _retval=$?
   209→
   210→    ignore rm "$_file"
   211→    ignore rmdir "$_dir"
   212→
   213→    return "$_retval"
   214→}
   215→
   216→get_current_exe() {
   217→    # Returns the executable used for system architecture detection
   218→    # This is only run on Linux
   219→    local _current_exe
   220→    if test -L /proc/self/exe ; then
   221→        _current_exe=/proc/self/exe
   222→    else
   223→        warn "Unable to find /proc/self/exe. System architecture detection might be inaccurate."
   224→        if test -n "$SHELL" ; then
   225→            _current_exe=$SHELL
   226→        else
   227→            need_cmd /bin/sh
   228→            _current_exe=/bin/sh
   229→        fi
   230→        warn "Falling back to $_current_exe."
   231→    fi
   232→    echo "$_current_exe"
   233→}
   234→
   235→get_bitness() {
   236→    need_cmd head
   237→    # Architecture detection without dependencies beyond coreutils.
   238→    # ELF files start out "\x7fELF", and the following byte is
   239→    #   0x01 for 32-bit and
   240→    #   0x02 for 64-bit.
   241→    # The printf builtin on some shells like dash only supports octal
   242→    # escape sequences, so we use those.
   243→    local _current_exe=$1
   244→    local _current_exe_head
   245→    _current_exe_head=$(head -c 5 "$_current_exe")
   246→    if [ "$_current_exe_head" = "$(printf '\177ELF\001')" ]; then
   247→        echo 32
   248→    elif [ "$_current_exe_head" = "$(printf '\177ELF\002')" ]; then
   249→        echo 64
   250→    else
   251→        err "unknown platform bitness"
   252→        exit 1;
   253→    fi
   254→}
   255→
   256→is_host_amd64_elf() {
   257→    local _current_exe=$1
   258→
   259→    need_cmd head
   260→    need_cmd tail
   261→    # ELF e_machine detection without dependencies beyond coreutils.
   262→    # Two-byte field at offset 0x12 indicates the CPU,
   263→    # but we're interested in it being 0x3E to indicate amd64, or not that.
   264→    local _current_exe_machine
   265→    _current_exe_machine=$(head -c 19 "$_current_exe" | tail -c 1)
   266→    [ "$_current_exe_machine" = "$(printf '\076')" ]
   267→}
   268→
   269→get_endianness() {
   270→    local _current_exe=$1
   271→    local cputype=$2
   272→    local suffix_eb=$3
   273→    local suffix_el=$4
   274→
   275→    # detect endianness without od/hexdump, like get_bitness() does.
   276→    need_cmd head
   277→    need_cmd tail
   278→
   279→    local _current_exe_endianness
   280→    _current_exe_endianness="$(head -c 6 "$_current_exe" | tail -c 1)"
   281→    if [ "$_current_exe_endianness" = "$(printf '\001')" ]; then
   282→        echo "${cputype}${suffix_el}"
   283→    elif [ "$_current_exe_endianness" = "$(printf '\002')" ]; then
   284→        echo "${cputype}${suffix_eb}"
   285→    else
   286→        err "unknown platform endianness"
   287→        exit 1
   288→    fi
   289→}
   290→
   291→# Detect the Linux/LoongArch UAPI flavor, with all errors being non-fatal.
   292→# Returns 0 or 234 in case of successful detection, 1 otherwise (/tmp being
   293→# noexec, or other causes).
   294→check_loongarch_uapi() {
   295→    need_cmd base64
   296→
   297→    local _tmp
   298→    if ! _tmp="$(ensure mktemp)"; then
   299→        return 1
   300→    fi
   301→
   302→    # Minimal Linux/LoongArch UAPI detection, exiting with 0 in case of
   303→    # upstream ("new world") UAPI, and 234 (-EINVAL truncated) in case of
   304→    # old-world (as deployed on several early commercial Linux distributions
   305→    # for LoongArch).
   306→    #
   307→    # See https://gist.github.com/xen0n/5ee04aaa6cecc5c7794b9a0c3b65fc7f for
   308→    # source to this helper binary.
   309→    ignore base64 -d > "$_tmp" <<EOF
   310→f0VMRgIBAQAAAAAAAAAAAAIAAgEBAAAAeAAgAAAAAABAAAAAAAAAAAAAAAAAAAAAQQAAAEAAOAAB
   311→AAAAAAAAAAEAAAAFAAAAAAAAAAAAAAAAACAAAAAAAAAAIAAAAAAAJAAAAAAAAAAkAAAAAAAAAAAA
   312→AQAAAAAABCiAAwUAFQAGABUAByCAAwsYggMAACsAC3iBAwAAKwAxen0n
   313→EOF
   314→
   315→    ignore chmod u+x "$_tmp"
   316→    if [ ! -x "$_tmp" ]; then
   317→        ignore rm "$_tmp"
   318→        return 1
   319→    fi
   320→
   321→    "$_tmp"
   322→    local _retval=$?
   323→
   324→    ignore rm "$_tmp"
   325→    return "$_retval"
   326→}
   327→
   328→ensure_loongarch_uapi() {
   329→    check_loongarch_uapi
   330→    case $? in
   331→        0)
   332→            return 0
   333→            ;;
   334→        234)
   335→            err 'Your Linux kernel does not provide the ABI required by this Rust distribution.'
   336→            err 'Please check with your OS provider for how to obtain a compatible Rust package for your system.'
   337→            exit 1
   338→            ;;
   339→        *)
   340→            warn "Cannot determine current system's ABI flavor, continuing anyway."
   341→            warn 'Note that the official Rust distribution only works with the upstream kernel ABI.'
   342→            warn 'Installation will fail if your running kernel happens to be incompatible.'
   343→            ;;
   344→    esac
   345→}

   347→get_architecture() {
   348→    local _ostype _cputype _bitness _arch _clibtype
   349→    _ostype="$(uname -s)"
   350→    _cputype="$(uname -m)"
   351→    _clibtype="gnu"
   352→
   353→    if [ "$_ostype" = Linux ]; then
   354→        if [ "$(uname -o)" = Android ]; then
   355→            _ostype=Android
   356→        fi
   357→        if ldd --version 2>&1 | grep -q 'musl'; then
   358→            _clibtype="musl"
   359→        fi
   360→    fi
   361→
   362→    if [ "$_ostype" = Darwin ]; then
   363→        # Darwin `uname -m` can lie due to Rosetta shenanigans. If you manage to
   364→        # invoke a native shell binary and then a native uname binary, you can
   365→        # get the real answer, but that's hard to ensure, so instead we use
   366→        # `sysctl` (which doesn't lie) to check for the actual architecture.
   367→        if [ "$_cputype" = i386 ]; then
   368→            # Handling i386 compatibility mode in older macOS versions (<10.15)
   369→            # running on x86_64-based Macs.
   370→            # Starting from 10.15, macOS explicitly bans all i386 binaries from running.
   371→            # See: <https://support.apple.com/en-us/HT208436>
   372→
   373→            # Avoid `sysctl: unknown oid` stderr output and/or non-zero exit code.
   374→            if (sysctl hw.optional.x86_64 2> /dev/null || true) | grep -q ': 1'; then
   375→                _cputype=x86_64
   376→            fi
   377→        elif [ "$_cputype" = x86_64 ]; then
   378→            # Handling x86-64 compatibility mode (a.k.a. Rosetta 2)
   379→            # in newer macOS versions (>=11) running on arm64-based Macs.
   380→            # Rosetta 2 is built exclusively for x86-64 and cannot run i386 binaries.
   381→
   382→            # Avoid `sysctl: unknown oid` stderr output and/or non-zero exit code.
   383→            if (sysctl hw.optional.arm64 2> /dev/null || true) | grep -q ': 1'; then
   384→                _cputype=arm64
   385→            fi
   386→        fi
   387→    fi
   388→
   389→    if [ "$_ostype" = SunOS ]; then
   390→        # Both Solaris and illumos presently announce as "SunOS" in "uname -s"
   391→        # so use "uname -o" to disambiguate.  We use the full path to the
   392→        # system uname in case the user has coreutils uname first in PATH,
   393→        # which has historically sometimes printed the wrong value here.
   394→        if [ "$(/usr/bin/uname -o)" = illumos ]; then
   395→            _ostype=illumos
   396→        fi
   397→
   398→        # illumos systems have multi-arch userlands, and "uname -m" reports the
   399→        # machine hardware name; e.g., "i86pc" on both 32- and 64-bit x86
   400→        # systems.  Check for the native (widest) instruction set on the
   401→        # running kernel:
   402→        if [ "$_cputype" = i86pc ]; then
   403→            _cputype="$(isainfo -n)"
   404→        fi
   405→    fi
   406→
   407→    local _current_exe
   408→    case "$_ostype" in
   409→
   410→        Android)
   411→            _ostype=linux-android
   412→            ;;
   413→
   414→        Linux)
   415→            _current_exe=$(get_current_exe)
   416→            _ostype=unknown-linux-$_clibtype
   417→            _bitness=$(get_bitness "$_current_exe")
   418→            ;;
   419→
   420→        FreeBSD)
   421→            _ostype=unknown-freebsd
   422→            ;;
   423→
   424→        NetBSD)
   425→            _ostype=unknown-netbsd
   426→            ;;
   427→
   428→        DragonFly)
   429→            _ostype=unknown-dragonfly
   430→            ;;
   431→
   432→        Darwin)
   433→            _ostype=apple-darwin
   434→            ;;
   435→
   436→        illumos)
   437→            _ostype=unknown-illumos
   438→            ;;
   439→
   440→        SunOS)
   441→            if [ "$_cputype" = sun4v ]; then
   442→               _ostype=sun-solaris
   443→            else
   444→               _ostype=pc-solaris
   445→            fi
   446→            ;;
   447→
   448→        MINGW* | MSYS* | CYGWIN* | Windows_NT)
   449→            _ostype=pc-windows-gnu
   450→            ;;
   451→
   452→        *)
   453→            err "unrecognized OS type: $_ostype"
   454→            exit 1
   455→            ;;
   456→
   457→    esac
   458→
   459→    case "$_cputype" in
   460→
   461→        i386 | i486 | i686 | i786 | x86)
   462→            _cputype=i686
   463→            ;;
   464→
   465→        xscale | arm)
   466→            _cputype=arm
   467→            if [ "$_ostype" = "linux-android" ]; then
   468→                _ostype=linux-androideabi
   469→            fi
   470→            ;;
   471→
   472→        armv6l)
   473→            _cputype=arm
   474→            if [ "$_ostype" = "linux-android" ]; then
   475→                _ostype=linux-androideabi
   476→            else
   477→                _ostype="${_ostype}eabihf"
   478→            fi
   479→            ;;
   480→
   481→        armv7l | armv8l)
   482→            _cputype=armv7
   483→            if [ "$_ostype" = "linux-android" ]; then
   484→                _ostype=linux-androideabi
   485→            else
   486→                _ostype="${_ostype}eabihf"
   487→            fi
   488→            ;;
   489→
   490→        aarch64 | arm64)
   491→            _cputype=aarch64
   492→            # Windows aarch64 uses gnullvm instead of gnu
   493→            if [ "$_ostype" = "pc-windows-gnu" ]; then
   494→                _ostype=pc-windows-gnullvm
   495→            fi
   496→            ;;
   497→
   498→        x86_64 | x86-64 | x64 | amd64)
   499→            _cputype=x86_64
   500→            ;;
   501→
   502→        mips)
   503→            _cputype=$(get_endianness "$_current_exe" mips '' el)
   504→            ;;
   505→
   506→        mips64)
   507→            if [ "$_bitness" -eq 64 ]; then
   508→                # only n64 ABI is supported for now
   509→                _ostype="${_ostype}abi64"
   510→                _cputype=$(get_endianness "$_current_exe" mips64 '' el)
   511→            fi
   512→            ;;
   513→
   514→        ppc)
   515→            _cputype=powerpc
   516→            ;;
   517→
   518→        ppc64)
   519→            _cputype=powerpc64
   520→            ;;
   521→
   522→        ppc64le)
   523→            _cputype=powerpc64le
   524→            ;;
   525→
   526→        s390x)
   527→            _cputype=s390x
   528→            ;;
   529→        sun4v)
   530→            _cputype=sparcv9
   531→            ;;
   532→        riscv64)
   533→            _cputype=riscv64gc
   534→            ;;
   535→        loongarch64)
   536→            _cputype=loongarch64
   537→            ensure_loongarch_uapi
   538→            ;;
   539→        *)
   540→            err "unknown CPU type: $_cputype"
   541→            exit 1
   542→
   543→    esac
   544→
   545→    # Detect 64-bit linux with 32-bit userland
   546→    if [ "${_ostype}" = unknown-linux-gnu ] && [ "${_bitness}" -eq 32 ]; then
   547→        case $_cputype in
   548→            x86_64)
   549→                if [ -n "${RUSTUP_CPUTYPE:-}" ]; then
   550→                    _cputype="$RUSTUP_CPUTYPE"
   551→                else {
   552→                    # 32-bit executable for amd64 = x32
   553→                    if is_host_amd64_elf "$_current_exe"; then {
   554→                        err "This host is running an x32 userland, for which no native toolchain is provided."
   555→                        err "You will have to install multiarch compatibility with i686 or amd64."
   556→                        err "To do so, set the RUSTUP_CPUTYPE environment variable set to i686 or amd64 and re-run this script."
   557→                        err "You will be able to add an x32 target after installation by running \`rustup target add x86_64-unknown-linux-gnux32\`."
   558→                         exit 1
   559→                    }; else
   560→                        _cputype=i686
   561→                    fi
   562→                }; fi
   563→                ;;
   564→            mips64)
   565→                _cputype=$(get_endianness "$_current_exe" mips '' el)
   566→                ;;
   567→            powerpc64)
   568→                _cputype=powerpc
   569→                ;;
   570→            aarch64)
   571→                _cputype=armv7
   572→                if [ "$_ostype" = "linux-android" ]; then
   573→                    _ostype=linux-androideabi
   574→                else
   575→                    _ostype="${_ostype}eabihf"
   576→                fi
   577→                ;;
   578→            riscv64gc)
   579→                err "riscv64 with 32-bit userland unsupported"
   580→                exit 1
   581→                ;;
   582→        esac
   583→    fi
   584→
   585→    # Detect armv7 but without the CPU features Rust needs in that build,
   586→    # and fall back to arm.
   587→    # See https://github.com/rust-lang/rustup.rs/issues/587.
   588→    if [ "$_ostype" = "unknown-linux-gnueabihf" ] && [ "$_cputype" = armv7 ]; then
   589→        if ! (ensure grep '^Features' /proc/cpuinfo | grep -E -q 'neon|simd') ; then
   590→            # Either `/proc/cpuinfo` is malformed or unavailable, or
   591→            # at least one processor does not have NEON (which is asimd on armv8+).
   592→            _cputype=arm
   593→        fi
   594→    fi
   595→
   596→    _arch="${_cputype}-${_ostype}"
   597→
   598→    RETVAL="$_arch"
   599→}
   600→
   601→__print() {
   602→    if $_ansi_escapes_are_valid; then
   603→        printf '\33[1m%s:\33[0m %s\n' "$1" "$2" >&2
   604→    else
   605→        printf '%s: %s\n' "$1" "$2" >&2
   606→    fi
   607→}
   608→
   609→warn() {
   610→    __print 'warn' "$1" >&2
   611→}
   612→
   613→say() {
   614→    if [ "$RUSTUP_QUIET" = "no" ]; then
   615→        __print 'info' "$1" >&2
   616→    fi
   617→}
   618→
   619→# NOTE: you are required to exit yourself
   620→# we don't do it here because of multiline errors
   621→err() {
   622→    __print 'error' "$1" >&2
   623→}
   624→
   625→need_cmd() {
   626→    if ! check_cmd "$1"; then
   627→        err "need '$1' (command not found)"
   628→        exit 1
   629→    fi

   660→# This wraps curl or wget. Try curl first, if not installed,
   661→# use wget instead.
   662→downloader() {
   663→    # zsh does not split words by default, Required for curl retry arguments below.
   664→    is_zsh && setopt local_options shwordsplit
   665→
   666→    local _dld
   667→    local _ciphersuites
   668→    local _err
   669→    local _status
   670→    local _retry
   671→    if check_cmd curl; then
   672→        # Check if we have a broken snap curl
   673→        # https://github.com/boukendesho/curl-snap/issues/1
   674→        _curl_path=$(command -v curl)
   675→        if echo "$_curl_path" | grep "/snap/" > /dev/null 2>&1; then
   676→            if check_cmd wget; then
   677→                _dld=wget
   678→            else
   679→                err "curl installed with snap cannot be used to install Rust"
   680→                err "due to missing permissions. Please uninstall it and"
   681→                err "reinstall curl with a different package manager (e.g., apt)."
   682→                err "See https://github.com/boukendesho/curl-snap/issues/1"
   683→                exit 1
   684→            fi
   685→        else
   686→            _dld=curl
   687→        fi
   688→    elif check_cmd wget; then
   689→        _dld=wget
   690→    else
   691→        _dld='curl or wget' # to be used in error message of need_cmd
   692→    fi
   693→
   694→    if [ "$1" = --check ]; then
   695→        need_cmd "$_dld"
   696→    elif [ "$_dld" = curl ]; then
   697→        check_curl_for_retry_support
   698→        _retry="$RETVAL"
   699→        get_ciphersuites_for_curl
   700→        _ciphersuites="$RETVAL"
   701→        if [ -n "$_ciphersuites" ]; then
   702→            # shellcheck disable=SC2086
   703→            _err=$(curl $_retry --proto '=https' --tlsv1.2 --ciphers "$_ciphersuites" --silent --show-error --fail --location "$1" --output "$2" 2>&1)
   704→            _status=$?
   705→        else
   706→            warn "Not enforcing strong cipher suites for TLS, this is potentially less secure"
   707→            if ! check_help_for "$3" curl --proto --tlsv1.2; then
   708→                warn "Not enforcing TLS v1.2, this is potentially less secure"
   709→                # shellcheck disable=SC2086
   710→                _err=$(curl $_retry --silent --show-error --fail --location "$1" --output "$2" 2>&1)
   711→                _status=$?
   712→            else
   713→                # shellcheck disable=SC2086
   714→                _err=$(curl $_retry --proto '=https' --tlsv1.2 --silent --show-error --fail --location "$1" --output "$2" 2>&1)
   715→                _status=$?
   716→            fi
   717→        fi
   718→        if [ -n "$_err" ]; then
   719→            warn "$_err"
   720→            if echo "$_err" | grep -q 404$; then
   721→                err "installer for platform '$3' not found, this may be unsupported"
   722→                exit 1
   723→            fi
   724→        fi
   725→        return $_status
   726→    elif [ "$_dld" = wget ]; then
   727→        if [ "$(wget -V 2>&1|head -2|tail -1|cut -f1 -d" ")" = "BusyBox" ]; then
   728→            warn "using the BusyBox version of wget.  Not enforcing strong cipher suites for TLS or TLS v1.2, this is potentially less secure"
   729→            _err=$(wget "$1" -O "$2" 2>&1)
   730→            _status=$?
   731→        else
   732→            get_ciphersuites_for_wget
   733→            _ciphersuites="$RETVAL"
   734→            if [ -n "$_ciphersuites" ]; then
   735→                _err=$(wget --https-only --secure-protocol=TLSv1_2 --ciphers "$_ciphersuites" "$1" -O "$2" 2>&1)
   736→                _status=$?
   737→            else
   738→                warn "Not enforcing strong cipher suites for TLS, this is potentially less secure"
   739→                if ! check_help_for "$3" wget --https-only --secure-protocol; then
   740→                    warn "Not enforcing TLS v1.2, this is potentially less secure"
   741→                    _err=$(wget "$1" -O "$2" 2>&1)
   742→                    _status=$?
   743→                else
   744→                    _err=$(wget --https-only --secure-protocol=TLSv1_2 "$1" -O "$2" 2>&1)
   745→                    _status=$?
   746→                fi
   747→            fi
   748→        fi
   749→        if [ -n "$_err" ]; then
   750→            warn "$_err"
   751→            if echo "$_err" | grep -q ' 404 Not Found$'; then
   752→                err "installer for platform '$3' not found, this may be unsupported"
   753→                exit 1
   754→            fi
   755→        fi
   756→        return $_status
   757→    else
   758→        err "Unknown downloader"   # should not reach here
   759→        exit 1
   760→    fi
   761→}

   763→check_help_for() {
   764→    local _arch
   765→    local _cmd
   766→    local _arg
   767→    _arch="$1"
   768→    shift
   769→    _cmd="$1"
   770→    shift
   771→
   772→    local _category
   773→    if "$_cmd" --help | grep -q '"--help all"'; then
   774→      _category="all"
   775→    else
   776→      _category=""
   777→    fi
   778→
   779→    case "$_arch" in
   780→
   781→        *darwin*)
   782→        if check_cmd sw_vers; then
   783→            local _os_version
   784→            local _os_major
   785→            _os_version=$(sw_vers -productVersion)
   786→            _os_major=$(echo "$_os_version" | cut -d. -f1)
   787→            case $_os_major in
   788→                10)
   789→                    # If we're running on macOS, older than 10.13, then we always
   790→                    # fail to find these options to force fallback
   791→                    if [ "$(echo "$_os_version" | cut -d. -f2)" -lt 13 ]; then
   792→                        # Older than 10.13
   793→                        warn "Detected macOS platform older than 10.13"
   794→                        return 1
   795→                    fi
   796→                    ;;
   797→                *)
   798→                    if ! { [ "$_os_major" -eq "$_os_major" ] 2>/dev/null && [ "$_os_major" -ge 11 ]; }; then
   799→                        # Unknown product version, warn and continue
   800→                        warn "Detected unknown macOS major version: $_os_version"
   801→                        warn "TLS capabilities detection may fail"
   802→                    fi
   803→                    ;; # We assume that macOS v11+ will always be okay.
   804→            esac
   805→        fi
   806→        ;;
   807→
   808→    esac
   809→
   810→    for _arg in "$@"; do
   811→        if ! "$_cmd" --help "$_category" | grep -q -- "$_arg"; then
   812→            return 1
   813→        fi
   814→    done
   815→
   816→    true # not strictly needed
   817→}

   819→# Check if curl supports the --retry flag, then pass it to the curl invocation.
   820→check_curl_for_retry_support() {
   821→    local _retry_supported=""
   822→    # "unspecified" is for arch, allows for possibility old OS using macports, homebrew, etc.
   823→    if check_help_for "notspecified" "curl" "--retry"; then
   824→        _retry_supported="--retry 3"
   825→        if check_help_for "notspecified" "curl" "--continue-at"; then
   826→            # "-C -" tells curl to automatically find where to resume the download when retrying.
   827→            _retry_supported="--retry 3 -C -"
   828→        fi
   829→    fi
   830→
   831→    RETVAL="$_retry_supported"
   832→}
   833→
   834→# Return cipher suite string specified by user, otherwise return strong TLS 1.2-1.3 cipher suites
   835→# if support by local tools is detected. Detection currently supports these curl backends:
   836→# GnuTLS and OpenSSL (possibly also LibreSSL and BoringSSL). Return value can be empty.
   837→get_ciphersuites_for_curl() {
   838→    if [ -n "${RUSTUP_TLS_CIPHERSUITES-}" ]; then
   839→        # user specified custom cipher suites, assume they know what they're doing
   840→        RETVAL="$RUSTUP_TLS_CIPHERSUITES"
   841→        return
   842→    fi
   843→
   844→    local _openssl_syntax="no"
   845→    local _gnutls_syntax="no"
   846→    local _backend_supported="yes"
   847→    if curl -V | grep -q ' OpenSSL/'; then
   848→        _openssl_syntax="yes"
   849→    elif curl -V | grep -iq ' LibreSSL/'; then
   850→        _openssl_syntax="yes"
   851→    elif curl -V | grep -iq ' BoringSSL/'; then
   852→        _openssl_syntax="yes"
   853→    elif curl -V | grep -iq ' GnuTLS/'; then
   854→        _gnutls_syntax="yes"
   855→    else
   856→        _backend_supported="no"
   857→    fi
   858→
   859→    local _args_supported="no"
   860→    if [ "$_backend_supported" = "yes" ]; then
   861→        # "unspecified" is for arch, allows for possibility old OS using macports, homebrew, etc.
   862→        if check_help_for "notspecified" "curl" "--tlsv1.2" "--ciphers" "--proto"; then
   863→            _args_supported="yes"
   864→        fi
   865→    fi
   866→
   867→    local _cs=""
   868→    if [ "$_args_supported" = "yes" ]; then
   869→        if [ "$_openssl_syntax" = "yes" ]; then
   870→            _cs=$(get_strong_ciphersuites_for "openssl")
   871→        elif [ "$_gnutls_syntax" = "yes" ]; then
   872→            _cs=$(get_strong_ciphersuites_for "gnutls")
   873→        fi
   874→    fi
   875→
   876→    RETVAL="$_cs"
   877→}
   878→
   879→# Return cipher suite string specified by user, otherwise return strong TLS 1.2-1.3 cipher suites
   880→# if support by local tools is detected. Detection currently supports these wget backends:
   881→# GnuTLS and OpenSSL (possibly also LibreSSL and BoringSSL). Return value can be empty.
   882→get_ciphersuites_for_wget() {
   883→    if [ -n "${RUSTUP_TLS_CIPHERSUITES-}" ]; then
   884→        # user specified custom cipher suites, assume they know what they're doing
   885→        RETVAL="$RUSTUP_TLS_CIPHERSUITES"
   886→        return
   887→    fi
   888→
   889→    local _cs=""
   890→    if wget -V | grep -q '\-DHAVE_LIBSSL'; then
   891→        # "unspecified" is for arch, allows for possibility old OS using macports, homebrew, etc.
   892→        if check_help_for "notspecified" "wget" "TLSv1_2" "--ciphers" "--https-only" "--secure-protocol"; then
   893→            _cs=$(get_strong_ciphersuites_for "openssl")
   894→        fi
   895→    elif wget -V | grep -q '\-DHAVE_LIBGNUTLS'; then
   896→        # "unspecified" is for arch, allows for possibility old OS using macports, homebrew, etc.
   897→        if check_help_for "notspecified" "wget" "TLSv1_2" "--ciphers" "--https-only" "--secure-protocol"; then
   898→            _cs=$(get_strong_ciphersuites_for "gnutls")
   899→        fi
   900→    fi
   901→
   902→    RETVAL="$_cs"
   903→}
   904→
   905→# Return strong TLS 1.2-1.3 cipher suites in OpenSSL or GnuTLS syntax. TLS 1.2
   906→# excludes non-ECDHE and non-AEAD cipher suites. DHE is excluded due to bad
   907→# DH params often found on servers (see RFC 7919). Sequence matches or is
   908→# similar to Firefox 68 ESR with weak cipher suites disabled via about:config.
   909→# $1 must be openssl or gnutls.
   910→get_strong_ciphersuites_for() {
   911→    if [ "$1" = "openssl" ]; then
   912→        # OpenSSL is forgiving of unknown values, no problems with TLS 1.3 values on versions that don't support it yet.
   913→        echo "TLS_AES_128_GCM_SHA256:TLS_CHACHA20_POLY1305_SHA256:TLS_AES_256_GCM_SHA384:ECDHE-ECDSA-AES128-GCM-SHA256:ECDHE-RSA-AES128-GCM-SHA256:ECDHE-ECDSA-CHACHA20-POLY1305:ECDHE-RSA-CHACHA20-POLY1305:ECDHE-ECDSA-AES256-GCM-SHA384:ECDHE-RSA-AES256-GCM-SHA384"
   914→    elif [ "$1" = "gnutls" ]; then
   915→        # GnuTLS isn't forgiving of unknown values, so this may require a GnuTLS version that supports TLS 1.3 even if wget doesn't.
   916→        # Begin with SECURE128 (and higher) then remove/add to build cipher suites. Produces same 9 cipher suites as OpenSSL but in slightly different order.
   917→        echo "SECURE128:-VERS-SSL3.0:-VERS-TLS1.0:-VERS-TLS1.1:-VERS-DTLS-ALL:-CIPHER-ALL:-MAC-ALL:-KX-ALL:+AEAD:+ECDHE-ECDSA:+ECDHE-RSA:+AES-128-GCM:+CHACHA20-POLY1305:+AES-256-GCM"
   918→    fi
   919→}
