    79→    79→pub fn material_def(name: &str) -> (String, f32) {
    80→    80→    let strength = match name {
    81→    81→        "led_strip" => 2.0,
    82→    82→        "screen_on" => 0.8,
    83→    83→        "terminal_screen" => 0.9,
    84→    84→        "exit_sign" => 1.4,
    85→    85→        "eyes" => 2.6,
    86→    86→        "light_panel" => 1.25,
    87→    87→        "battery" => 0.35,
    88→    88→        "receipt_paper" => 0.05,
    89→    89→        "poster_a" | "poster_b" | "poster_c" | "poster_d" => 0.04,
    90→    90→        "entity_mask" => 0.03,
    91→    91→        _ => 0.0,
    92→    92→    };
    93→    93→    (name.to_string(), strength)
    94→    94→}

   148→   148→    pub fn new(window: std::sync::Arc<Window>) -> Renderer {
   149→   149→        let size = window.inner_size();
   150→   150→        let instance = wgpu::Instance::new(wgpu::InstanceDescriptor::default());
   151→   151→        let surface = instance
   152→   152→            .create_surface(window.clone())
   153→   153→            .expect("création surface");
   154→   154→        let adapter = pollster::block_on(instance.request_adapter(&wgpu::RequestAdapterOptions {
   155→   155→            power_preference: wgpu::PowerPreference::HighPerformance,
   156→   156→            compatible_surface: Some(&surface),
   157→   157→            force_fallback_adapter: false,
   158→   158→        }))
   159→   159→        .expect("pas de GPU compatible (Vulkan/Metal/DX12 requis)");
   160→   160→
   161→   161→        let (device, queue) = pollster::block_on(adapter.request_device(
   162→   162→            &wgpu::DeviceDescriptor {
   163→   163→                label: Some("sl3-device"),
   164→   164→                required_features: wgpu::Features::empty(),
   165→   165→                required_limits: wgpu::Limits::default(),
   166→   166→                memory_hints: wgpu::MemoryHints::Performance,
   167→   167→            },
   168→   168→            None,
   169→   169→        ))
   170→   170→        .expect("création device");
   171→   171→
   172→   172→        let caps = surface.get_capabilities(&adapter);
   173→   173→        let format = caps
   174→   174→            .formats
   175→   175→            .iter()
   176→   176→            .find(|f| f.is_srgb())
   177→   177→            .copied()
   178→   178→            .unwrap_or(caps.formats[0]);
   179→   179→        let surface_config = wgpu::SurfaceConfiguration {
   180→   180→            usage: wgpu::TextureUsages::RENDER_ATTACHMENT,
   181→   181→            format,
   182→   182→            width: size.width.max(1),
   183→   183→            height: size.height.max(1),
   184→   184→            present_mode: wgpu::PresentMode::Fifo,
   185→   185→            alpha_mode: caps.alpha_modes[0],
   186→   186→            view_formats: vec![],
   187→   187→            desired_maximum_frame_latency: 2,
   188→   188→        };
   189→   189→        surface.configure(&device, &surface_config);
   190→   190→
   191→   191→        // Modèles + matériaux.
   192→   192→        let models = model::load_all_models(&device);
   193→   193→        let mut material_bind_groups = HashMap::new();
   194→   194→        let mat_layout = Self::material_layout(&device);
   195→   195→        for name in [
   196→   196→            "concrete", "concrete_dark", "floor_hall", "floor_corridor", "floor_office",
   197→   197→            "floor_server", "floor_arch", "floor_elec", "ceiling", "metal_dark",
   198→   198→            "rack_front", "server_front", "led_strip", "screen_off", "screen_on",
   199→   199→            "terminal_screen", "desk_wood", "terminal_body", "chair_fabric", "door_metal",
   200→   200→            "door_frame", "shelf_metal", "cardboard", "entity_cloth", "entity_mask",
   201→   201→            "eyes", "tech_vest", "tech_pants", "tech_head", "receipt_paper",
   202→   202→            "breaker_panel", "exit_sign", "battery", "extinguisher", "duct",
   203→   203→            "light_panel", "hazard", "poster_a", "poster_b", "poster_c", "poster_d",
   204→   204→        ] {
   205→   205→            let (file, strength) = material_def(name);
   206→   206→            let path = Path::new("assets/textures").join(format!("{file}.png"));
   207→   207→            let tex = texture::load_png(&device, &queue, &path);
   208→   208→            let mat_buf = device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
   209→   209→                label: None,
   210→   210→                contents: bytemuck::cast_slice(&[1.0f32, 1.0, 1.0, strength]),
   211→   211→                usage: wgpu::BufferUsages::UNIFORM | wgpu::BufferUsages::COPY_DST,
   212→   212→            });
   213→   213→            let bg = device.create_bind_group(&wgpu::BindGroupDescriptor {
   214→   214→                label: Some(name),
   215→   215→                layout: &mat_layout,
   216→   216→                entries: &[
   217→   217→                    wgpu::BindGroupEntry { binding: 0, resource: wgpu::BindingResource::Sampler(&Self::world_sampler(&device)) },
   218→   218→                    wgpu::BindGroupEntry { binding: 1, resource: wgpu::BindingResource::TextureView(&tex.view) },
   219→   219→                    wgpu::BindGroupEntry { binding: 2, resource: mat_buf.as_entire_binding() },
   220→   220→                ],
   221→   221→            });
   222→   222→            material_bind_groups.insert(name.to_string(), bg);
   223→   223→        }
   224→   224→
   225→   225→        // Uniforms monde.
   226→   226→        let world_uniform_buf = device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
   227→   227→            label: Some("world-uniform"),
   228→   228→            contents: bytemuck::bytes_of(&WorldUniform {
   229→   229→                view_proj: glam::Mat4::IDENTITY.to_cols_array_2d(),
   230→   230→                cam_pos: [0.0; 4],
   231→   231→                light_pos: [[0.0; 4]; MAX_LIGHTS],
   232→   232→                light_col: [[0.0; 4]; MAX_LIGHTS],
   233→   233→                flash_pos: [0.0; 4],
   234→   234→                flash_dir: [0.0; 4],
   235→   235→                misc: [0.0; 4],
   236→   236→                flash_col: [0.0; 4],
   237→   237→            }),
   238→   238→            usage: wgpu::BufferUsages::UNIFORM | wgpu::BufferUsages::COPY_DST,
   239→   239→        });
   240→   240→        let world_bind_layout = Self::world_bind_layout(&device);
   241→   241→        let world_bind0 = device.create_bind_group(&wgpu::BindGroupDescriptor {
   242→   242→            label: Some("world-bind0"),
   243→   243→            layout: &world_bind_layout,
   244→   244→            entries: &[wgpu::BindGroupEntry {
   245→   245→                binding: 0,
   246→   246→                resource: world_uniform_buf.as_entire_binding(),
   247→   247→            }],
   248→   248→        });
   249→   249→
   250→   250→        let world_shader = device.create_shader_module(wgpu::ShaderModuleDescriptor {
   251→   251→            label: Some("world.wgsl"),
   252→   252→            source: wgpu::ShaderSource::Wgsl(include_str!("../shaders/world.wgsl").into()),
   253→   253→        });
   254→   254→        let world_pipeline = Self::world_pipeline(&device, &world_bind_layout, &mat_layout, &world_shader, format);
   255→   255→
   256→   256→        // Post-process.
   257→   257→        let post_shader = device.create_shader_module(wgpu::ShaderModuleDescriptor {
   258→   258→            label: Some("post.wgsl"),
   259→   259→            source: wgpu::ShaderSource::Wgsl(include_str!("../shaders/post.wgsl").into()),
   260→   260→        });
   261→   261→        let post_layout = device.create_bind_group_layout(&wgpu::BindGroupLayoutDescriptor {
   262→   262→            label: Some("post-layout"),
   263→   263→            entries: &[
   264→   264→                wgpu::BindGroupLayoutEntry { binding: 0, visibility: wgpu::ShaderStages::FRAGMENT, ty: wgpu::BindingType::Buffer { ty: wgpu::BufferBindingType::Uniform, has_dynamic_offset: false, min_binding_size: None }, count: None },
   265→   265→                wgpu::BindGroupLayoutEntry { binding: 1, visibility: wgpu::ShaderStages::FRAGMENT, ty: wgpu::BindingType::Sampler(wgpu::SamplerBindingType::Filtering), count: None },
   266→   266→                wgpu::BindGroupLayoutEntry { binding: 2, visibility: wgpu::ShaderStages::FRAGMENT, ty: wgpu::BindingType::Texture { sample_type: wgpu::TextureSampleType::Float { filterable: true }, multisampled: false, view_dimension: wgpu::TextureViewDimension::D2 }, count: None },
   267→   267→            ],
   268→   268→        });
   269→   269→        let post_uniform_buf = device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
   270→   270→            label: Some("post-uniform"),
   271→   271→            contents: bytemuck::cast_slice(&[0.0f32; 4]),
   272→   272→            usage: wgpu::BufferUsages::UNIFORM | wgpu::BufferUsages::COPY_DST,
   273→   273→        });
   274→   274→
   275→   275→        let (offscreen_tex, offscreen_view) = Self::make_offscreen(&device, size.width.max(1), size.height.max(1), format);
   276→   276→        let (depth_tex, depth_view) = Self::make_depth(&device, size.width.max(1), size.height.max(1));
   277→   277→        let post_bind = device.create_bind_group(&wgpu::BindGroupDescriptor {
   278→   278→            label: Some("post-bind"),
   279→   279→            layout: &post_layout,
   280→   280→            entries: &[
   281→   281→                wgpu::BindGroupEntry { binding: 0, resource: post_uniform_buf.as_entire_binding() },
   282→   282→                wgpu::BindGroupEntry { binding: 1, resource: wgpu::BindingResource::Sampler(&texture::linear_sampler(&device, false)) },
   283→   283→                wgpu::BindGroupEntry { binding: 2, resource: wgpu::BindingResource::TextureView(&offscreen_view) },
   284→   284→            ],
   285→   285→        });
   286→   286→        let post_pipeline = Self::post_pipeline(&device, &post_layout, &post_shader, format);
   287→   287→
   288→   288→        // UI.
   289→   289→        let ui_shader = device.create_shader_module(wgpu::ShaderModuleDescriptor {
   290→   290→            label: Some("ui.wgsl"),
   291→   291→            source: wgpu::ShaderSource::Wgsl(include_str!("../shaders/ui.wgsl").into()),

   303→   303→            label: Some("ui-layout1"),

   305→   305→                wgpu::BindGroupLayoutEntry { binding: 0, visibility: wgpu::ShaderStages::FRAGMENT, ty: wgpu::BindingType::Sampler(wgpu::SamplerBindingType::Filtering), count: None },

   549→   549→    pub fn resize(&mut self, w: u32, h: u32) {
   550→   550→        let (w, h) = (w.max(1), h.max(1));
   551→   551→        self.surface_config.width = w;
   552→   552→        self.surface_config.height = h;
   553→   553→        self.surface.configure(&self.device, &self.surface_config);
   554→   554→        let (sw, sh) = self.scaled_size();
   555→   555→        let (t, v) = Self::make_offscreen(&self.device, sw, sh, self.surface_config.format);
   556→   556→        self.offscreen_tex = t;
   557→   557→        self.offscreen_view = v;
   558→   558→        self.rebuild_post_bind();
   559→   559→        let (dt, dv) = Self::make_depth(&self.device, sw, sh);
   560→   560→        self.depth_tex = dt;
   561→   561→        self.depth_view = dv;
   562→   562→        let ui_data = [w as f32, h as f32, 0.0, 0.0];
   563→   563→        self.queue.write_buffer(&self.ui_uniform_buf, 0, bytemuck::cast_slice(&ui_data));
   564→   564→    }

   587→   587→    /// Change l'échelle de rendu (recrée offscreen + depth, post/UI restent en pleine résolution).
   588→   588→    pub fn set_render_scale(&mut self, scale: f32) {
   589→   589→        let s = scale.clamp(0.45, 1.0);
   590→   590→        if (s - self.render_scale).abs() < 0.001 {
   591→   591→            return;
   592→   592→        }
   593→   593→        self.render_scale = s;
   594→   594→        let (sw, sh) = self.scaled_size();
   595→   595→        let (t, v) = Self::make_offscreen(&self.device, sw, sh, self.surface_config.format);
   596→   596→        self.offscreen_tex = t;
   597→   597→        self.offscreen_view = v;
   598→   598→        self.rebuild_post_bind();
   599→   599→        let (dt, dv) = Self::make_depth(&self.device, sw, sh);
   600→   600→        self.depth_tex = dt;
   601→   601→        self.depth_view = dv;
   602→   602→    }

   605→   605→        self.render_scale
   606→   606→    }

   608→   608→    // ---------- construction des batchs statiques ----------
   609→   609→    pub fn build_static(&self, insts: &[(String, InstanceData)]) -> StaticBatches {
   610→   610→        let mut groups: HashMap<PartKey, Vec<InstanceRaw>> = HashMap::new();
   611→   611→        for (model_name, inst) in insts {
   612→   612→            if let Some(model) = self.models.get(model_name) {
   613→   613→                for (pi, part) in model.parts.iter().enumerate() {
   614→   614→                    groups
   615→   615→                        .entry(PartKey { model: model_name.clone(), part: pi })
   616→   616→                        .or_default()
   617→   617→                        .push(inst.raw());
   618→   618→                    let _ = &part.mat;
   619→   619→                }
   620→   620→            }
   621→   621→        }
   622→   622→        let mut batches = Vec::new();
   623→   623→        let mut total = 0usize;
   624→   624→        for (key, raws) in groups {
   625→   625→            let buffer = self.device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
   626→   626→                label: Some("static-instances"),
   627→   627→                contents: bytemuck::cast_slice(&raws),
   628→   628→                usage: wgpu::BufferUsages::VERTEX | wgpu::BufferUsages::COPY_DST,
   629→   629→            });
   630→   630→            total += raws.len();
   631→   631→            batches.push(StaticBatch { key, buffer, count: raws.len() as u32 });
   632→   632→        }
   633→   633→        StaticBatches { batches, total_instances: total }
   634→   634→    }

   636→   636→    // ---------- dessin ----------
   637→   637→    pub fn draw_instances_pass(
   638→   638→        &mut self,
   639→   639→        pass: &mut wgpu::RenderPass<'static>,
   640→   640→        statics: &StaticBatches,
   641→   641→        dynamics: &[(String, Vec<InstanceData>)],
   642→   642→    ) {
   643→   643→        pass.set_pipeline(&self.world_pipeline);
   644→   644→        pass.set_bind_group(0, &self.world_bind0, &[]);
   645→   645→
   646→   646→        // Statiques.
   647→   647→        for batch in &statics.batches {
   648→   648→            self.draw_part(pass, &batch.key, &batch.buffer, batch.count);
   649→   649→        }
   650→   650→
   651→   651→        // Dynamiques.
   652→   652→        for (model_name, insts) in dynamics {
   653→   653→            let Some(model) = self.models.get(model_name) else { continue };
   654→   654→            for (pi, _part) in model.parts.iter().enumerate() {
   655→   655→                if insts.is_empty() {
   656→   656→                    continue;
   657→   657→                }
   658→   658→                let key = PartKey { model: model_name.clone(), part: pi };
   659→   659→                let raws: Vec<InstanceRaw> = insts.iter().map(|i| i.raw()).collect();
   660→   660→                let bytes_needed = (raws.len() * std::mem::size_of::<InstanceRaw>()) as u64;
   661→   661→                let needs_grow = self
   662→   662→                    .dyn_scratch
   663→   663→                    .get(&key)
   664→   664→                    .map(|(_, cap)| *cap < bytes_needed)
   665→   665→                    .unwrap_or(true);
   666→   666→                if needs_grow {
   667→   667→                    let cap = bytes_needed.max(1024);
   668→   668→                    let buf = self.device.create_buffer(&wgpu::BufferDescriptor {
   669→   669→                        label: Some("dyn-instances"),
   670→   670→                        size: cap,
   671→   671→                        usage: wgpu::BufferUsages::VERTEX | wgpu::BufferUsages::COPY_DST,
   672→   672→                        mapped_at_creation: false,
   673→   673→                    });
   674→   674→                    self.dyn_scratch.insert(key.clone(), (buf, cap));
   675→   675→                }
   676→   676→                let (buf, _cap) = &self.dyn_scratch[&key];
   677→   677→                self.queue.write_buffer(buf, 0, bytemuck::cast_slice(&raws));
   678→   678→                self.draw_part(pass, &key, buf, raws.len() as u32);
   679→   679→            }
   680→   680→        }
   681→   681→    }
   682→   682→
   683→   683→    fn draw_part(
   684→   684→        &self,
   685→   685→        pass: &mut wgpu::RenderPass<'static>,
   686→   686→        key: &PartKey,
   687→   687→        instance_buf: &wgpu::Buffer,
   688→   688→        count: u32,
   689→   689→    ) {
   690→   690→        let Some(model) = self.models.get(&key.model) else { return };
   691→   691→        let Some(part) = model.parts.get(key.part) else { return };
   692→   692→        let Some(bg) = self.material_bind_groups.get(&part.mat) else { return };
   693→   693→        pass.set_bind_group(1, bg, &[]);
   694→   694→        pass.set_vertex_buffer(0, part.vertex_buf.slice(..));
   695→   695→        pass.set_vertex_buffer(1, instance_buf.slice(..));
   696→   696→        pass.set_index_buffer(part.index_buf.slice(..), wgpu::IndexFormat::Uint32);
   697→   697→        pass.draw_indexed(0..part.count, 0, 0..count);
   698→   698→    }

   700→   700→    #[allow(clippy::too_many_arguments)]
   701→   701→    pub fn render(
   702→   702→        &mut self,
   703→   703→        world_uniform: &WorldUniform,
   704→   704→        statics: &StaticBatches,
   705→   705→        dynamics: &[(String, Vec<InstanceData>)],
   706→   706→        ui_ops: &[ui::UiOp],
   707→   707→        post_params: [f32; 4],
   708→   708→    ) -> Result<(), wgpu::SurfaceError> {
   709→   709→        let frame = self.surface.get_current_texture()?;
   710→   710→        let frame_view = frame.texture.create_view(&wgpu::TextureViewDescriptor::default());
   711→   711→
   712→   712→        self.queue
   713→   713→            .write_buffer(&self.world_uniform_buf, 0, bytemuck::bytes_of(world_uniform));
   714→   714→        self.queue
   715→   715→            .write_buffer(&self.post_uniform_buf, 0, bytemuck::cast_slice(&post_params));
   716→   716→
   717→   717→        // Quads UI.
   718→   718→        let (verts, draws) = ui::build_ui_verts(ui_ops, &self.font);
   719→   719→        if !verts.is_empty() {
   720→   720→            let bytes_needed = (verts.len() * std::mem::size_of::<ui::UiVertex>()) as u64;
   721→   721→            if self.ui_buf.size() < bytes_needed {
   722→   722→                self.ui_buf = self.device.create_buffer(&wgpu::BufferDescriptor {
   723→   723→                    label: Some("ui-verts"),
   724→   724→                    size: bytes_needed * 2,
   725→   725→                    usage: wgpu::BufferUsages::VERTEX | wgpu::BufferUsages::COPY_DST,
   726→   726→                    mapped_at_creation: false,
   727→   727→                });
   728→   728→            }
   729→   729→            self.queue.write_buffer(&self.ui_buf, 0, bytemuck::cast_slice(&verts));
   730→   730→        }
   731→   731→
   732→   732→        let mut encoder = self.device.create_command_encoder(&wgpu::CommandEncoderDescriptor {
   733→   733→            label: Some("frame"),
   734→   734→        });
   735→   735→
   736→   736→        // 1) Monde -> offscreen.
   737→   737→        {
   738→   738→            let pass = encoder.begin_render_pass(&wgpu::RenderPassDescriptor {
   739→   739→                label: Some("world-pass"),
   740→   740→                color_attachments: &[Some(wgpu::RenderPassColorAttachment {
   741→   741→                    view: &self.offscreen_view,
   742→   742→                    resolve_target: None,
   743→   743→                    ops: wgpu::Operations {
   744→   744→                        load: wgpu::LoadOp::Clear(wgpu::Color {
   745→   745→                            r: 0.004,
   746→   746→                            g: 0.005,
   747→   747→                            b: 0.008,
   748→   748→                            a: 1.0,
   749→   749→                        }),
   750→   750→                        store: wgpu::StoreOp::Store,
   751→   751→                    },
   752→   752→                })],
   753→   753→                depth_stencil_attachment: Some(wgpu::RenderPassDepthStencilAttachment {
   754→   754→                    view: &self.depth_view,
   755→   755→                    depth_ops: Some(wgpu::Operations {
   756→   756→                        load: wgpu::LoadOp::Clear(1.0),
   757→   757→                        store: wgpu::StoreOp::Store,
   758→   758→                    }),
   759→   759→                    stencil_ops: None,
   760→   760→                }),
   761→   761→                timestamp_writes: None,
   762→   762→                occlusion_query_set: None,
   763→   763→            });
   764→   764→            // 'pass borrow self...' : étendre la durée de vie via transmute sûr ici
   765→   765→            let mut pass = pass.forget_lifetime();
   766→   766→            self.draw_instances_pass(&mut pass, statics, dynamics);
   767→   767→        }
   768→   768→
   769→   769→        // 2) Post-process -> surface.
   770→   770→        {
   771→   771→            let mut pass = encoder.begin_render_pass(&wgpu::RenderPassDescriptor {
   772→   772→                label: Some("post-pass"),
   773→   773→                color_attachments: &[Some(wgpu::RenderPassColorAttachment {
   774→   774→                    view: &frame_view,
   775→   775→                    resolve_target: None,
   776→   776→                    ops: wgpu::Operations {
   777→   777→                        load: wgpu::LoadOp::Clear(wgpu::Color::BLACK),
   778→   778→                        store: wgpu::StoreOp::Store,
   779→   779→                    },
   780→   780→                })],
   781→   781→                depth_stencil_attachment: None,
   782→   782→                timestamp_writes: None,
   783→   783→                occlusion_query_set: None,
   784→   784→            });
   785→   785→            pass.set_pipeline(&self.post_pipeline);
   786→   786→            pass.set_bind_group(0, &self.post_bind, &[]);
   787→   787→            pass.draw(0..3, 0..1);
   788→   788→        }
   789→   789→
   790→   790→        // 3) UI -> surface.
   791→   791→        if !verts.is_empty() {
   792→   792→            let mut pass = encoder.begin_render_pass(&wgpu::RenderPassDescriptor {
   793→   793→                label: Some("ui-pass"),
   794→   794→                color_attachments: &[Some(wgpu::RenderPassColorAttachment {
   795→   795→                    view: &frame_view,
   796→   796→                    resolve_target: None,
   797→   797→                    ops: wgpu::Operations {
   798→   798→                        load: wgpu::LoadOp::Load,
   799→   799→                        store: wgpu::StoreOp::Store,
   800→   800→                    },
   801→   801→                })],
   802→   802→                depth_stencil_attachment: None,
   803→   803→                timestamp_writes: None,
   804→   804→                occlusion_query_set: None,
   805→   805→            });
   806→   806→            pass.set_pipeline(&self.ui_pipeline);
   807→   807→            pass.set_bind_group(0, &self.ui_bind0, &[]);
   808→   808→            let stride = std::mem::size_of::<ui::UiVertex>() as u64;
   809→   809→            let mut cursor = 0u64;
   810→   810→            for (_op, (is_text, quads)) in ui_ops.iter().zip(draws.iter()) {
   811→   811→                if *quads == 0 {
   812→   812→                    continue;
   813→   813→                }
   814→   814→                match is_text {
   815→   815→                    true => pass.set_bind_group(1, &self.font_bg, &[]),
   816→   816→                    false => pass.set_bind_group(1, &self.white_bg, &[]),
   817→   817→                }
   818→   818→                let bytes = *quads as u64 * 4 * stride;
   819→   819→                pass.set_vertex_buffer(0, self.ui_buf.slice(cursor..cursor + bytes));
   820→   820→                pass.draw(0..4, 0..*quads);
   821→   821→                cursor += bytes;
   822→   822→            }
   823→   823→        }
   824→   824→
   825→   825→        self.queue.submit(Some(encoder.finish()));
   826→   826→        frame.present();
   827→   827→        Ok(())
   828→   828→    }
   829→   829→}
   830→   830→
