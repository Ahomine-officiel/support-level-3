    79→    79→pub fn material_def(name: &str) -> (String, f32) {
    80→    80→    let strength = match name {
    81→    81→        "led_strip" => 2.0,
    82→    82→        "screen_on" => 0.8,
    83→    83→        "terminal_screen" => 0.9,
    84→    84→        "exit_sign" => 1.4,
    85→    85→        "eyes" => 2.6,
    86→    86→        "light_panel" => 1.25,
    87→    87→        "battery" => 0.35,
    88→    88→        "receipt_paper" => 0.05,
    89→    89→        "poster_a" | "poster_b" | "poster_c" | "poster_d" => 0.04,
    90→    90→        "entity_mask" => 0.03,
    91→    91→        _ => 0.0,
    92→    92→    };
    93→    93→    (name.to_string(), strength)
    94→    94→}

   547→   547→    pub fn resize(&mut self, w: u32, h: u32) {
   548→   548→        let (w, h) = (w.max(1), h.max(1));
   549→   549→        self.surface_config.width = w;
   550→   550→        self.surface_config.height = h;
   551→   551→        self.surface.configure(&self.device, &self.surface_config);
   552→   552→        let (t, v) = Self::make_offscreen(&self.device, w, h, self.surface_config.format);
   553→   553→        self.offscreen_tex = t;
   554→   554→        self.offscreen_view = v;
   555→   555→        let post_layout = self.post_pipeline.get_bind_group_layout(0);
   556→   556→        self.post_bind = self.device.create_bind_group(&wgpu::BindGroupDescriptor {
   557→   557→            label: Some("post-bind"),
   558→   558→            layout: &post_layout,
   559→   559→            entries: &[
   560→   560→                wgpu::BindGroupEntry { binding: 0, resource: self.post_uniform_buf.as_entire_binding() },
   561→   561→                wgpu::BindGroupEntry { binding: 1, resource: wgpu::BindingResource::Sampler(&texture::linear_sampler(&self.device, false)) },
   562→   562→                wgpu::BindGroupEntry { binding: 2, resource: wgpu::BindingResource::TextureView(&self.offscreen_view) },
   563→   563→            ],
   564→   564→        });
   565→   565→        let (dt, dv) = Self::make_depth(&self.device, w, h);
   566→   566→        self.depth_tex = dt;
   567→   567→        self.depth_view = dv;
   568→   568→        let ui_data = [w as f32, h as f32, 0.0, 0.0];
   569→   569→        self.queue.write_buffer(&self.ui_uniform_buf, 0, bytemuck::cast_slice(&ui_data));
   570→   570→    }

   572→   572→    // ---------- construction des batchs statiques ----------
   573→   573→    pub fn build_static(&self, insts: &[(String, InstanceData)]) -> StaticBatches {
   574→   574→        let mut groups: HashMap<PartKey, Vec<InstanceRaw>> = HashMap::new();
   575→   575→        for (model_name, inst) in insts {
   576→   576→            if let Some(model) = self.models.get(model_name) {
   577→   577→                for (pi, part) in model.parts.iter().enumerate() {
   578→   578→                    groups
   579→   579→                        .entry(PartKey { model: model_name.clone(), part: pi })
   580→   580→                        .or_default()
   581→   581→                        .push(inst.raw());
   582→   582→                    let _ = &part.mat;
   583→   583→                }
   584→   584→            }
   585→   585→        }
   586→   586→        let mut batches = Vec::new();
   587→   587→        let mut total = 0usize;
   588→   588→        for (key, raws) in groups {
   589→   589→            let buffer = self.device.create_buffer_init(&wgpu::util::BufferInitDescriptor {
   590→   590→                label: Some("static-instances"),
   591→   591→                contents: bytemuck::cast_slice(&raws),
   592→   592→                usage: wgpu::BufferUsages::VERTEX | wgpu::BufferUsages::COPY_DST,
   593→   593→            });
   594→   594→            total += raws.len();
   595→   595→            batches.push(StaticBatch { key, buffer, count: raws.len() as u32 });
   596→   596→        }
   597→   597→        StaticBatches { batches, total_instances: total }
   598→   598→    }

   601→   601→    pub fn draw_instances_pass(
   602→   602→        &mut self,
   603→   603→        pass: &mut wgpu::RenderPass<'static>,
   604→   604→        statics: &StaticBatches,
   605→   605→        dynamics: &[(String, Vec<InstanceData>)],
   606→   606→    ) {
   607→   607→        pass.set_pipeline(&self.world_pipeline);
   608→   608→        pass.set_bind_group(0, &self.world_bind0, &[]);
   609→   609→
   610→   610→        // Statiques.
   611→   611→        for batch in &statics.batches {
   612→   612→            self.draw_part(pass, &batch.key, &batch.buffer, batch.count);
   613→   613→        }
   614→   614→
   615→   615→        // Dynamiques.
   616→   616→        for (model_name, insts) in dynamics {
   617→   617→            let Some(model) = self.models.get(model_name) else { continue };
   618→   618→            for (pi, _part) in model.parts.iter().enumerate() {
   619→   619→                if insts.is_empty() {
   620→   620→                    continue;
   621→   621→                }
   622→   622→                let key = PartKey { model: model_name.clone(), part: pi };
   623→   623→                let raws: Vec<InstanceRaw> = insts.iter().map(|i| i.raw()).collect();
   624→   624→                let bytes_needed = (raws.len() * std::mem::size_of::<InstanceRaw>()) as u64;
   625→   625→                let needs_grow = self
   626→   626→                    .dyn_scratch
   627→   627→                    .get(&key)
   628→   628→                    .map(|(_, cap)| *cap < bytes_needed)
   629→   629→                    .unwrap_or(true);
   630→   630→                if needs_grow {
   631→   631→                    let cap = bytes_needed.max(1024);
   632→   632→                    let buf = self.device.create_buffer(&wgpu::BufferDescriptor {
   633→   633→                        label: Some("dyn-instances"),
   634→   634→                        size: cap,
   635→   635→                        usage: wgpu::BufferUsages::VERTEX | wgpu::BufferUsages::COPY_DST,
   636→   636→                        mapped_at_creation: false,
   637→   637→                    });
   638→   638→                    self.dyn_scratch.insert(key.clone(), (buf, cap));
   639→   639→                }
   640→   640→                let (buf, _cap) = &self.dyn_scratch[&key];
   641→   641→                self.queue.write_buffer(buf, 0, bytemuck::cast_slice(&raws));
   642→   642→                self.draw_part(pass, &key, buf, raws.len() as u32);
   643→   643→            }
   644→   644→        }
   645→   645→    }
   646→   646→
   647→   647→    fn draw_part(
   648→   648→        &self,
   649→   649→        pass: &mut wgpu::RenderPass<'static>,
   650→   650→        key: &PartKey,
   651→   651→        instance_buf: &wgpu::Buffer,
   652→   652→        count: u32,
   653→   653→    ) {
   654→   654→        let Some(model) = self.models.get(&key.model) else { return };
   655→   655→        let Some(part) = model.parts.get(key.part) else { return };
   656→   656→        let Some(bg) = self.material_bind_groups.get(&part.mat) else { return };
   657→   657→        pass.set_bind_group(1, bg, &[]);
   658→   658→        pass.set_vertex_buffer(0, part.vertex_buf.slice(..));
   659→   659→        pass.set_vertex_buffer(1, instance_buf.slice(..));
   660→   660→        pass.set_index_buffer(part.index_buf.slice(..), wgpu::IndexFormat::Uint32);
   661→   661→        pass.draw_indexed(0..part.count, 0, 0..count);
   662→   662→    }

   664→   664→    #[allow(clippy::too_many_arguments)]
   665→   665→    pub fn render(
   666→   666→        &mut self,
   667→   667→        world_uniform: &WorldUniform,
   668→   668→        statics: &StaticBatches,
   669→   669→        dynamics: &[(String, Vec<InstanceData>)],
   670→   670→        ui_ops: &[ui::UiOp],
   671→   671→        post_params: [f32; 4],
   672→   672→    ) -> Result<(), wgpu::SurfaceError> {
   673→   673→        let frame = self.surface.get_current_texture()?;
   674→   674→        let frame_view = frame.texture.create_view(&wgpu::TextureViewDescriptor::default());
   675→   675→
   676→   676→        self.queue
   677→   677→            .write_buffer(&self.world_uniform_buf, 0, bytemuck::bytes_of(world_uniform));
   678→   678→        self.queue
   679→   679→            .write_buffer(&self.post_uniform_buf, 0, bytemuck::cast_slice(&post_params));
   680→   680→
   681→   681→        // Quads UI.
   682→   682→        let (verts, draws) = ui::build_ui_verts(ui_ops, &self.font);
   683→   683→        if !verts.is_empty() {
   684→   684→            let bytes_needed = (verts.len() * std::mem::size_of::<ui::UiVertex>()) as u64;
   685→   685→            if self.ui_buf.size() < bytes_needed {
   686→   686→                self.ui_buf = self.device.create_buffer(&wgpu::BufferDescriptor {
   687→   687→                    label: Some("ui-verts"),
   688→   688→                    size: bytes_needed * 2,
   689→   689→                    usage: wgpu::BufferUsages::VERTEX | wgpu::BufferUsages::COPY_DST,
   690→   690→                    mapped_at_creation: false,
   691→   691→                });
   692→   692→            }
   693→   693→            self.queue.write_buffer(&self.ui_buf, 0, bytemuck::cast_slice(&verts));
   694→   694→        }
   695→   695→
   696→   696→        let mut encoder = self.device.create_command_encoder(&wgpu::CommandEncoderDescriptor {
   697→   697→            label: Some("frame"),
   698→   698→        });
   699→   699→
   700→   700→        // 1) Monde -> offscreen.
   701→   701→        {
   702→   702→            let pass = encoder.begin_render_pass(&wgpu::RenderPassDescriptor {
   703→   703→                label: Some("world-pass"),
   704→   704→                color_attachments: &[Some(wgpu::RenderPassColorAttachment {
   705→   705→                    view: &self.offscreen_view,
   706→   706→                    resolve_target: None,
   707→   707→                    ops: wgpu::Operations {
   708→   708→                        load: wgpu::LoadOp::Clear(wgpu::Color {
   709→   709→                            r: 0.004,
   710→   710→                            g: 0.005,
   711→   711→                            b: 0.008,
   712→   712→                            a: 1.0,
   713→   713→                        }),
   714→   714→                        store: wgpu::StoreOp::Store,
   715→   715→                    },
   716→   716→                })],
   717→   717→                depth_stencil_attachment: Some(wgpu::RenderPassDepthStencilAttachment {
   718→   718→                    view: &self.depth_view,
   719→   719→                    depth_ops: Some(wgpu::Operations {
   720→   720→                        load: wgpu::LoadOp::Clear(1.0),
   721→   721→                        store: wgpu::StoreOp::Store,
   722→   722→                    }),
   723→   723→                    stencil_ops: None,
   724→   724→                }),
   725→   725→                timestamp_writes: None,
   726→   726→                occlusion_query_set: None,
   727→   727→            });
   728→   728→            // 'pass borrow self...' : étendre la durée de vie via transmute sûr ici
   729→   729→            let mut pass = pass.forget_lifetime();
   730→   730→            self.draw_instances_pass(&mut pass, statics, dynamics);
   731→   731→        }
   732→   732→
   733→   733→        // 2) Post-process -> surface.
   734→   734→        {
   735→   735→            let mut pass = encoder.begin_render_pass(&wgpu::RenderPassDescriptor {
   736→   736→                label: Some("post-pass"),
   737→   737→                color_attachments: &[Some(wgpu::RenderPassColorAttachment {
   738→   738→                    view: &frame_view,
   739→   739→                    resolve_target: None,
   740→   740→                    ops: wgpu::Operations {
   741→   741→                        load: wgpu::LoadOp::Clear(wgpu::Color::BLACK),
   742→   742→                        store: wgpu::StoreOp::Store,
   743→   743→                    },
   744→   744→                })],
   745→   745→                depth_stencil_attachment: None,
   746→   746→                timestamp_writes: None,
   747→   747→                occlusion_query_set: None,
   748→   748→            });
   749→   749→            pass.set_pipeline(&self.post_pipeline);
   750→   750→            pass.set_bind_group(0, &self.post_bind, &[]);
   751→   751→            pass.draw(0..3, 0..1);
   752→   752→        }
   753→   753→
   754→   754→        // 3) UI -> surface.
   755→   755→        if !verts.is_empty() {
   756→   756→            let mut pass = encoder.begin_render_pass(&wgpu::RenderPassDescriptor {
   757→   757→                label: Some("ui-pass"),
   758→   758→                color_attachments: &[Some(wgpu::RenderPassColorAttachment {
   759→   759→                    view: &frame_view,
   760→   760→                    resolve_target: None,
   761→   761→                    ops: wgpu::Operations {
   762→   762→                        load: wgpu::LoadOp::Load,
   763→   763→                        store: wgpu::StoreOp::Store,
   764→   764→                    },
   765→   765→                })],
   766→   766→                depth_stencil_attachment: None,
   767→   767→                timestamp_writes: None,
   768→   768→                occlusion_query_set: None,
   769→   769→            });
   770→   770→            pass.set_pipeline(&self.ui_pipeline);
   771→   771→            pass.set_bind_group(0, &self.ui_bind0, &[]);
   772→   772→            let stride = std::mem::size_of::<ui::UiVertex>() as u64;
   773→   773→            let mut cursor = 0u64;
   774→   774→            for (_op, (is_text, quads)) in ui_ops.iter().zip(draws.iter()) {
   775→   775→                if *quads == 0 {
   776→   776→                    continue;
   777→   777→                }
   778→   778→                match is_text {
   779→   779→                    true => pass.set_bind_group(1, &self.font_bg, &[]),
   780→   780→                    false => pass.set_bind_group(1, &self.white_bg, &[]),
   781→   781→                }
   782→   782→                let bytes = *quads as u64 * 4 * stride;
   783→   783→                pass.set_vertex_buffer(0, self.ui_buf.slice(cursor..cursor + bytes));
   784→   784→                pass.draw(0..4, 0..*quads);
   785→   785→                cursor += bytes;
   786→   786→            }
   787→   787→        }
   788→   788→
   789→   789→        self.queue.submit(Some(encoder.finish()));
   790→   790→        frame.present();
   791→   791→        Ok(())
   792→   792→    }
   793→   793→}
